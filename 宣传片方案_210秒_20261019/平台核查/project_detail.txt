              >
                <p :data-i18n-skip="template.description ? true : undefined">{{ template.description || '暂无描述' }}</p>
              </el-tooltip>

        @size-change="changeProjectTemplatePageSize"

<script setup>
import { computed, onMounted, reactive, ref, watch } from 'vue'
import { useRoute, useRouter } from 'vue-router'
import { ElMessage, ElMessageBox } from 'element-plus'
import {
  Clock,
  CopyDocument,
  Delete,
  Document,
  DocumentCopy,
  Edit,
  Plus,
  Search,
  View,
  VideoPlay,
} from '@element-plus/icons-vue'

import { useProjectStore } from '@/stores/projects'
import { useTemplateStore } from '@/stores/templates'
import { translateKnownText } from '@/i18n'
import { useAppStore } from '@/stores/app'
import { copyTextToClipboard } from '@/utils/clipboard'
import { formatDate } from '@/utils/format'
import { createTemplate } from '@/api/templates'
import {
  buildTemplatePayload,
  buildTimedTemplateName,
  buildWorkflowRoute,
  getCreatedTemplateId,
  getWorkflowId,
} from '@/utils/templateExecution'
import WorkflowConfigDrawer from './components/WorkflowConfigDrawer.vue'
import WorkflowYamlPreviewDialog from './components/WorkflowYamlPreviewDialog.vue'

const route = useRoute()
const router = useRouter()
const appStore = useAppStore()
const projectStore = useProjectStore()
const templateStore = useTemplateStore()

const projectAction = reactive({
  executingTemplateId: '',
  deletingTemplateId: '',
  previewingTemplateId: '',
})

const templatePreview = reactive({
  visible: false,
  definition: {},
})

const copyTemplateDialog = reactive({
  visible: false,
  loading: false,
  template: null,
  projectId: '',
  name: '',
})

// 任务配置抽屉
const workflowDrawerVisible = ref(false)
const workflowDrawerRef = ref(null)
const loadedProjectTemplateProjectId = ref('')
const projectTemplateLoadSerial = ref(0)
const projectRouteLoadSerial = ref(0)
const projectRouteLoading = ref(true)

const activeProjectId = computed(() => String(route.params.projectId || route.query.project_id || route.query.projectId || '').trim())
const activeProject = computed(() => {
  if (!activeProjectId.value) {
    return null
  }

  return projectStore.projects.find((project) => String(project.id) === activeProjectId.value) || projectStore.current
})
const projectTemplateFilterReady = computed(() => (
  loadedProjectTemplateProjectId.value === activeProjectId.value &&
  templateStore.filters.projectId === activeProjectId.value
))
const projectDetailLoading = computed(() => (
  projectRouteLoading.value || projectStore.detailLoading || templateStore.loading
))
const visibleProjectTemplates = computed(() => (
  projectTemplateFilterReady.value ? templateStore.templates : []
))
const canShowTemplateEmpty = computed(() => (
  projectTemplateFilterReady.value && !templateStore.loading && !visibleProjectTemplates.value.length
))
const projectTemplateKeyword = computed({
  get: () => templateStore.filters.keyword,
  set: (value) => {
    templateStore.filters.keyword = String(value || '')
  },
})

function getTemplateTags(template, limit = 3) {
  return Array.isArray(template?.tags) ? template.tags.slice(0, limit) : []
}

function translateKnown(value) {
  return translateKnownText(value, appStore.language)
}

async function copyId(id) {
  const value = String(id || '').trim()

  if (!value) {
    ElMessage.warning('暂无可复制的 ID')
    return
  }

  try {
    await copyTextToClipboard(value)
    ElMessage.success('ID 已复制')
  } catch (error) {
    ElMessage.error(error.message || '复制失败')
  }
}

async function searchProjectTemplates() {
  await templateStore.loadTemplates({
    project_id: activeProjectId.value,
    is_builtin: undefined,
    offset: 0,
  })
}

function openProjectTemplates() {
  workflowDrawerVisible.value = true
}

function openTemplateWorkflowConfig(template, options = {}) {
  if (!template?.id) {
    return
  }

  workflowDrawerVisible.value = true
  workflowDrawerRef.value?.openWithTemplate(template, options)
}

function isPlainRecord(value) {
  return value !== null && typeof value === 'object' && !Array.isArray(value)
}

async function executeProjectTemplate(template) {
  if (!template?.id) {
    ElMessage.warning('模板 ID 缺失，无法执行')
    return
  }

  try {
    projectAction.executingTemplateId = template.id
    const projectId = activeProjectId.value
    const result = await templateStore.execute(template.id, {
      name: buildTimedTemplateName(template.name),
      parameters: isPlainRecord(template.parameters) ? template.parameters : {},
      configs: isPlainRecord(template.configs) ? template.configs : {},
      project_id: projectId || undefined,
    })
    const workflowId = getWorkflowId(result)

    ElMessage.success(result?.message || '模板已执行')
    router.push(buildWorkflowRoute(workflowId, projectId, {
      templateId: template.id,
      preferList: true,
    }))
  } catch (error) {
    if (!error?.response) {
      ElMessage.error(error.message || '模板执行失败')
    }
  } finally {
    projectAction.executingTemplateId = ''
  }
}

function openTemplateWorkflowDetail(template) {
  router.push(buildWorkflowRoute('', activeProjectId.value, {
    templateId: template?.id,
  }))
}

async function getFullTemplate(template) {
  if (!template?.id) {
    return null
  }

  if (template.definition) {
    return template
  }

  await templateStore.selectTemplate(template.id)
  return templateStore.current
}

async function openTemplatePreview(template) {
  if (!template?.id) {
    ElMessage.warning('模板 ID 缺失，无法预览')
    return
  }

  try {
    projectAction.previewingTemplateId = template.id
    const fullTemplate = await getFullTemplate(template)
    templatePreview.definition = fullTemplate?.definition || {}
    templatePreview.visible = true
  } catch (error) {
    if (!error?.response) {
      ElMessage.error(error.message || '模板预览加载失败')
    }
  } finally {
    projectAction.previewingTemplateId = ''
  }
}

function openCopyTemplateDialog(template) {
  if (!template?.id) {
    ElMessage.warning('模板 ID 缺失，无法复制')
    return
  }

  copyTemplateDialog.template = template
  copyTemplateDialog.projectId = activeProjectId.value || ''
  copyTemplateDialog.name = buildTimedTemplateName(template.name || '未命名模板')
  copyTemplateDialog.visible = true

  if (!projectStore.projects.length && !projectStore.loading) {
    projectStore.loadProjects().catch(() => {})
  }
}

async function confirmCopyTemplate() {
  const targetProjectId = String(copyTemplateDialog.projectId || '').trim()
  const templateName = String(copyTemplateDialog.name || '').trim()

  if (!targetProjectId) {
    ElMessage.warning('请选择项目')
    return
  }

  if (!templateName) {
    ElMessage.warning('请输入模板名称')
    return
  }

  try {
    copyTemplateDialog.loading = true
    const sourceTemplate = await getFullTemplate(copyTemplateDialog.template)

    if (!sourceTemplate?.definition) {
      throw new Error('模板内容缺失，无法复制')
    }

    const result = await createTemplate({
      ...buildTemplatePayload(sourceTemplate, targetProjectId),
      name: templateName,
    })
    const createdId = getCreatedTemplateId(result)

    copyTemplateDialog.visible = false
    ElMessage.success('模板已复制')

    if (targetProjectId === activeProjectId.value) {
      await reloadProjectTemplates()
    } else if (createdId) {
      await templateStore.loadTemplates({
        project_id: activeProjectId.value,
        is_builtin: undefined,
      })
    }
  } catch (error) {
    if (!error?.response) {
      ElMessage.error(error.message || '模板复制失败')
    }
  } finally {
    copyTemplateDialog.loading = false
  }
}

async function removeProjectTemplate(template) {
  if (!template?.id) {
    return
  }

  const confirmed = await ElMessageBox.confirm(`确认删除模板「${template.name || template.id}」？`, '删除模板', {
    type: 'warning',
  })
    .then(() => true)
    .catch(() => false)

  if (!confirmed) {
    return
  }

  try {
    projectAction.deletingTemplateId = template.id
    await templateStore.remove(template.id)
    await reloadProjectTemplates()
    ElMessage.success('模板已删除')
  } finally {
    projectAction.deletingTemplateId = ''
  }
}

function openTemplateEdit(template) {
  if (!template?.id) {
    return
  }

  openTemplateWorkflowConfig(template, { mode: 'edit' })
}

async function loadProjectTemplates(projectId, options = {}) {
  if (!projectId) {
    loadedProjectTemplateProjectId.value = ''
    return
  }

  const { resetPage = true } = options
  const nextOffset = resetPage ? 0 : templateStore.pagination.offset
  const loadSerial = projectTemplateLoadSerial.value + 1

  projectTemplateLoadSerial.value = loadSerial

  if (loadedProjectTemplateProjectId.value !== projectId) {
    loadedProjectTemplateProjectId.value = ''
  }

  templateStore.filters.projectId = projectId
  templateStore.filters.keyword = ''
  templateStore.filters.builtin = ''
  templateStore.pagination.offset = nextOffset

  await templateStore.loadTemplates({
    project_id: projectId,
    is_builtin: undefined,
    offset: nextOffset,
  })

  if (loadSerial === projectTemplateLoadSerial.value && activeProjectId.value === projectId) {
    loadedProjectTemplateProjectId.value = projectId
  }
}

async function reloadProjectTemplates() {
  await loadProjectTemplates(activeProjectId.value, { resetPage: false })
}

async function handleWorkflowCreated(result) {
  // 刷新模板列表
  await reloadProjectTemplates()
}

async function changeProjectTemplatePage(page) {
  await templateStore.loadTemplates({
    project_id: activeProjectId.value,
    is_builtin: undefined,
    offset: (page - 1) * templateStore.pagination.limit,
  })
}

async function changeProjectTemplatePageSize(size) {
  await templateStore.loadTemplates({
    project_id: activeProjectId.value,
    is_builtin: undefined,
    limit: size,
    offset: 0,
  })
}

async function syncProjectRoute(projectId) {
  const routeLoadSerial = projectRouteLoadSerial.value + 1

  projectRouteLoadSerial.value = routeLoadSerial
  projectRouteLoading.value = true
  loadedProjectTemplateProjectId.value = ''

  try {
    if (!projectId) {
      projectStore.current = null
      templateStore.setProjectFilter('')
      return
    }

    const localProject = projectStore.projects.find((project) => String(project.id) === projectId)

    if (localProject) {
      projectStore.current = localProject
    } else {
      await projectStore.selectProject(projectId)
    }

    await loadProjectTemplates(projectId)
  } finally {
    if (routeLoadSerial === projectRouteLoadSerial.value) {
      projectRouteLoading.value = false
    }
  }
}

onMounted(async () => {
  await projectStore.loadProjects()
  await syncProjectRoute(activeProjectId.value)
})

watch(activeProjectId, syncProjectRoute)
</script>
