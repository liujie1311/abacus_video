          </div>
        </div>
        <div
          v-if="canShowRetryWorkflow(workflowStore.current) || canStopWorkflow(workflowStore.current)"
          class="workflow-detail-actions"
        >
          <el-button
            v-if="canShowRetryWorkflow(workflowStore.current)"
            plain
            :disabled="!canRetryWorkflow(workflowStore.current)"
            :icon="RefreshRight"
            @click="retry(workflowStore.current)"
          >
            重试
          </el-button>
          <el-button
            v-if="canStopWorkflow(workflowStore.current)"
            type="warning"
            plain
            :icon="VideoPause"
            @click="stop(workflowStore.current)"
          >
            停止
          </el-button>
        </div>
      </section>

      <div
        v-if="workflowStore.current.error"
        class="workflow-error-alert"
        role="alert"
      >
        <span class="workflow-error-icon">
          <el-icon><CircleCloseFilled /></el-icon>
        </span>
        <div class="workflow-error-copy">
          <strong>工作流错误</strong>
          <p data-i18n-skip>{{ workflowStore.current.error }}</p>
        </div>
      </div>

      <div class="workflow-detail-grid">
        <section class="detail-panel task-panel">
          <div class="panel-head">
            <div>
              <h2>任务详情</h2>
              <p>点击任务查看日志</p>
            </div>
          </div>

          <div v-if="workflowStore.tasks.length" class="task-detail-list">
            <article
              v-for="(task, index) in workflowStore.tasks"
              :key="getTaskId(task) || `${getTaskStepId(task)}-${index}`"
              class="task-detail-card"
              :class="[
                `status-${normalizeStatus(task.status)}`,
                { active: getTaskId(task) === selectedTaskId },
              ]"
              @click="selectTask(task)"
            >
              <span class="task-index">{{ index + 1 }}</span>
              <span class="task-state-emblem" :title="workflowStatusText(task.status)">
                <el-icon>
                  <component :is="getStatusIcon(task.status)" />
                </el-icon>
              </span>
              <div class="task-copy">
                <div class="task-title-row">
                  <strong>{{ getTaskStepId(task) }}</strong>
                  <span v-if="getTaskType(task) !== '-'" class="task-type-tag">{{ getTaskType(task) }}</span>
                </div>
                <div v-if="getTaskRetryCount(task) > 0" class="task-meta-row">
                  <span v-if="getTaskRetryCount(task) > 0">重试 {{ getTaskRetryCount(task) }}</span>
                </div>
                <div v-if="getTaskId(task)" class="task-id-line" :title="getTaskId(task)">
                  <span class="task-id-label">ID:</span>
                  <span class="task-id-value">{{ getTaskId(task) }}</span>
                  <el-button
                    link
                    type="primary"
                    size="small"
                    :icon="CopyDocument"
                    class="task-inline-copy"
                    @click.stop="copyTaskId(task)"
                  >
                    复制
                  </el-button>
                </div>
                <div v-if="getTaskTimeEntries(task).length" class="task-time-row">
                  <span v-for="entry in getTaskTimeEntries(task)" :key="entry.key" :title="formatDate(entry.value)">
                    <span>{{ entry.label }}</span>
                    <strong>{{ formatDate(entry.value) }}</strong>
                  </span>
                </div>
                <div v-if="getTaskCopyablePaths(task).length" class="task-output-list">
                  <div v-for="entry in getTaskCopyablePaths(task)" :key="entry.key" class="task-output-row" :title="entry.path">
                    <span class="task-output-label">{{ entry.label }}</span>
                    <button
                      v-if="isBrowsableTaskPathEntry(entry)"
                      type="button"
                      class="task-output-path task-output-path-link"
                      :title="entry.path"
                      data-i18n-skip
                      @click.stop="openHostFileBrowser(entry.path)"
                    >
                      {{ entry.path }}
                    </button>
                    <span v-else class="task-output-path" :title="entry.path" data-i18n-skip>{{ entry.path }}</span>
                    <el-button link type="primary" :icon="CopyDocument" @click.stop="copyTaskPath(entry.path)">
                      复制
                    </el-button>
                  </div>
                </div>
                <p v-if="task.error" class="task-error" data-i18n-skip>{{ task.error }}</p>
              </div>
            </article>
          </div>
          <div v-else class="empty-block compact">暂无任务步骤</div>
        </section>

        <section class="detail-panel log-panel">
          <div class="panel-head">
            <div>
              <h2>任务日志</h2>
              <p>{{ selectedTaskId || '请选择任务' }}</p>
            </div>
          </div>

          <pre class="code-preview workflow-log-preview">{{ logText }}</pre>
        </section>
      </div>

      <HostFileBrowserDialog
        v-model:visible="hostFileBrowserVisible"
        :initial-path="hostFileBrowserInitialPath"
      />
    </template>
  </div>

<script setup>
import { computed, onBeforeUnmount, onMounted, ref, watch } from 'vue'
import { useRoute, useRouter } from 'vue-router'
import { ElMessage, ElMessageBox } from 'element-plus'
import {
  CircleCheckFilled,
  CircleCloseFilled,
  Clock,
  CopyDocument,
  Loading,
  RefreshRight,
  VideoPause,
  WarningFilled,
} from '@element-plus/icons-vue'

import { getWorkflowTaskLogs } from '@/api/workflows'
import HostFileBrowserDialog from '@/views/host-files/HostFileBrowserDialog.vue'
import { useEventStore } from '@/stores/events'
import { useWorkflowStore } from '@/stores/workflows'
import { copyTextToClipboard } from '@/utils/clipboard'
import { formatDate, formatJson, workflowStatusText } from '@/utils/format'
import { createRealtimeLogEntry, isSameRealtimeLog } from '@/utils/workflowRealtime'

const route = useRoute()
const router = useRouter()
const eventStore = useEventStore()
const workflowStore = useWorkflowStore()
const selectedTaskId = ref('')
const taskLogs = ref([])
const hostFileBrowserVisible = ref(false)
const hostFileBrowserInitialPath = ref('')
const statusIcons = {
  cancelled: CircleCloseFilled,
  completed: CircleCheckFilled,
  failed: WarningFilled,
  paused: VideoPause,
  pending: Clock,
  running: Loading,
  skipped: CircleCloseFilled,
}
const realtimeRefreshDelay = 800
let currentReloadTimer = 0
let taskLogReloadTimer = 0
let taskLogRequestSeq = 0
let currentReloadWorkflowId = ''
let taskLogReloadTarget = {
  workflowId: '',
  taskId: '',
}

const workflowId = computed(() => String(route.params.workflowId || ''))
const routeProjectId = computed(() => String(route.query.project_id || route.query.projectId || '').trim())
const routeTemplateId = computed(() => String(route.query.template_id || route.query.templateId || '').trim())
function getRawLogText(item) {
  if (typeof item === 'string') {
    return item
  }

  if (!item) {
    return ''
  }

  const rawText = item.message ?? item.content ?? item.text ?? item.line
  return rawText === undefined || rawText === null ? formatJson(item) : String(rawText)
}

const logText = computed(() => {
  const logs = taskLogs.value

  if (!logs?.length) {
    return '暂无日志'
  }

  return logs
    .map(getRawLogText)
    .filter((line) => line !== '')
    .join('\n')
})
const latestRealtimeEventId = computed(() => eventStore.latestEvent?.id || '')

async function loadWorkflowDetail() {
  if (!workflowId.value) {
    router.replace({
      path: '/workflows',
      query: buildWorkflowListQuery(),
    })
    return
  }

  selectedTaskId.value = ''
  taskLogs.value = []

  try {
    await workflowStore.selectWorkflow(workflowId.value)
    await selectFirstTask()
  } catch {
    router.replace({
      path: '/workflows',
      query: buildWorkflowListQuery(),
    })
  }
}

async function ensureTaskSelection() {
  if (
    selectedTaskId.value &&
    workflowStore.tasks.some((task) => getTaskId(task) === selectedTaskId.value)
  ) {
    await loadTaskLogsById(selectedTaskId.value)
    return
  }

  await selectFirstTask()
}

async function selectFirstTask() {
  const firstTask = workflowStore.tasks.find((task) => getTaskId(task))

  if (!firstTask) {
    selectedTaskId.value = ''
    taskLogs.value = []
    return
  }

  selectedTaskId.value = getTaskId(firstTask)
  await loadTaskLogsById(selectedTaskId.value)
}

function buildWorkflowListQuery() {
  return {
    ...(routeProjectId.value ? { project_id: routeProjectId.value } : {}),
    ...(routeTemplateId.value ? { template_id: routeTemplateId.value } : {}),
  }
}

async function selectTask(task) {
  const taskId = getTaskId(task)

  if (!taskId) {
    ElMessage.warning('任务 ID 缺失')
    return
  }

  selectedTaskId.value = taskId
  await loadTaskLogsById(taskId)
}

async function loadTaskLogsById(taskId) {
  if (!workflowId.value || !taskId) {
    return
  }

  const requestSeq = ++taskLogRequestSeq

  taskLogs.value = []
  const logs = await getWorkflowTaskLogs(workflowId.value, taskId)

  if (requestSeq === taskLogRequestSeq && selectedTaskId.value === taskId) {
    taskLogs.value = logs
  }
}

function openHostFileBrowser(path) {
  const text = String(path || '').trim()

  if (!text) {
    ElMessage.warning('该任务暂无可浏览的目录')
    return
  }

  hostFileBrowserInitialPath.value = text
  hostFileBrowserVisible.value = true
}

async function retry(workflow) {
  if (!canRetryWorkflow(workflow)) {
    return
  }

  await workflowStore.retry(workflow.id)
  ElMessage.success('已提交重试')
}

async function stop(workflow) {
  if (!canStopWorkflow(workflow)) {
    return
  }

  const confirmed = await ElMessageBox.confirm(`确认停止工作流「${workflow.name}」？`, '停止工作流', {
    type: 'warning',
  })
    .then(() => true)
    .catch(() => false)

  if (!confirmed) {
    return
  }

  await workflowStore.stop(workflow.id)
  ElMessage.success('已停止')
}

async function copyTaskPath(path) {
  if (!path) {
    ElMessage.warning('该任务暂无可复制的目录')
    return
  }

  try {
    await copyTextToClipboard(path)
    ElMessage.success('路径已复制')
  } catch {
    ElMessage.error('复制失败，请手动选择内容复制')
  }
}

async function copyTaskId(task) {
  const taskId = getTaskId(task)

  if (!taskId) {
    ElMessage.warning('该任务暂无可复制的 ID')
    return
  }

  try {
    await copyTextToClipboard(taskId)
    ElMessage.success('任务 ID 已复制')
  } catch {
    ElMessage.error('复制失败，请手动选择内容复制')
  }
}

function normalizeStatus(status) {
  return String(status || 'unknown')
    .trim()
    .toLowerCase()
    .replace(/[^a-z0-9_-]+/g, '-')
}

function getStatusIcon(status) {
  return statusIcons[normalizeStatus(status)] || Clock
}

function getFinishTimeLabel(status) {
  if (status === 'failed') {
    return '失败'
  }

  if (status === 'cancelled') {
    return '取消'
  }

  return '完成'
}

function canRetryWorkflow(workflow) {
  return Boolean(workflow?.id) && normalizeStatus(workflow?.status) !== 'running' && canShowRetryWorkflow(workflow)
}

function canShowRetryWorkflow(workflow) {
  return normalizeStatus(workflow?.status) !== 'completed'
}

function canStopWorkflow(workflow) {
  return normalizeStatus(workflow?.status) === 'running'
}

function getTaskId(task) {
  return task?.id || task?.task_id || ''
}

function getTaskStepId(task) {
  return task?.step_id || task?.stepId || getTaskId(task) || '-'
}

function getTaskType(task) {
  return task?.type || task?.task_type || '-'
}

function getTaskRetryCount(task) {
  return Math.max(0, Number(task?.retry_count || task?.retryCount || task?.retries || 0) || 0)
}

function getTaskTimeEntries(task) {
  if (!task) {
    return []
  }

  const status = normalizeStatus(task.status)
  const finalTime =
    task.completed_at ||
    task.finished_at ||
    task.ended_at ||
    task.end_time ||
    (['completed', 'failed', 'cancelled'].includes(status) ? task.updated_at : '')
  const entries = [
    { key: 'created_at', label: '创建', value: task.created_at },
    { key: 'started_at', label: '开始', value: task.started_at },
    { key: 'completed_at', label: getFinishTimeLabel(status), value: finalTime },
  ]

  if (!finalTime && task.updated_at && task.updated_at !== task.created_at) {
    entries.push({ key: 'updated_at', label: '更新', value: task.updated_at })
  }

  return entries.filter((entry) => Boolean(entry.value))
}

function getTaskCopyablePaths(task) {
  const outputs = task?.outputs && typeof task.outputs === 'object' ? task.outputs : {}
  return ['workdir', 'output_dir']
    .map((key) => ({
      key,
      label: getTaskPathLabel(key),
      path: String(outputs[key] || '').trim(),
    }))
    .filter((entry) => isPathLikeValue(entry.path))
}

function getTaskPathLabel(key) {
  const labels = {
    workdir: '工作目录',
    output_dir: '输出目录',
  }

  return labels[key] || key
}

function isBrowsableTaskPathEntry(entry) {
  return ['workdir', 'output_dir'].includes(entry?.key || '')
}

function isPathLikeValue(value) {
  if (typeof value !== 'string') {
    return false
  }

  const text = value.trim()
  return /^([A-Za-z]:[\\/]|\/|\\\\)/.test(text)
}

function appendSelectedTaskLog(event) {
  const logEntry = createRealtimeLogEntry(event)

  if (!logEntry || taskLogs.value.some((log) => isSameRealtimeLog(log, logEntry))) {
    return
  }

  taskLogs.value = [...taskLogs.value, logEntry]
}

function scheduleCurrentReload(reloadWorkflowId) {
  if (!reloadWorkflowId || workflowStore.current?.id !== reloadWorkflowId) {
    return
  }

  currentReloadWorkflowId = reloadWorkflowId

  if (currentReloadTimer) {
    return
  }

  currentReloadTimer = window.setTimeout(async () => {
    const targetWorkflowId = currentReloadWorkflowId

    currentReloadTimer = 0
    currentReloadWorkflowId = ''

    if (workflowStore.current?.id !== targetWorkflowId) {
      return
    }

    try {
      await workflowStore.reloadCurrent()
      await ensureTaskSelection()
    } catch {
      // 请求拦截器会统一展示错误，这里只避免定时任务产生未处理异常。
    }
  }, realtimeRefreshDelay)
}

function scheduleSelectedTaskLogsReload(reloadWorkflowId, taskId) {
  if (!reloadWorkflowId || !taskId) {
    return
  }

  taskLogReloadTarget = {
    workflowId: reloadWorkflowId,
    taskId,
  }

  if (taskLogReloadTimer) {
    return
  }

  taskLogReloadTimer = window.setTimeout(async () => {
    const reloadTarget = taskLogReloadTarget

    taskLogReloadTimer = 0
    taskLogReloadTarget = {
      workflowId: '',
      taskId: '',
    }

    if (workflowStore.current?.id !== reloadTarget.workflowId || selectedTaskId.value !== reloadTarget.taskId) {
      return
    }

    try {
      await loadTaskLogsById(reloadTarget.taskId)
    } catch {
      // 请求拦截器会统一展示错误，这里只避免定时任务产生未处理异常。
    }
  }, realtimeRefreshDelay)
}

function handleRealtimeEvent(event) {
  const result = workflowStore.applyRealtimeEvent(event)

  if (!result?.workflowId || result.workflowId !== workflowId.value) {
    return
  }

  scheduleCurrentReload(result.workflowId)

  if (selectedTaskId.value && result.taskId === selectedTaskId.value) {
    appendSelectedTaskLog(event)
    scheduleSelectedTaskLogsReload(result.workflowId, result.taskId)
  }
}

onMounted(async () => {
  eventStore.connect()
  await loadWorkflowDetail()
})

watch(workflowId, loadWorkflowDetail)
watch(latestRealtimeEventId, () => handleRealtimeEvent(eventStore.latestEvent))

onBeforeUnmount(() => {
  window.clearTimeout(currentReloadTimer)
  window.clearTimeout(taskLogReloadTimer)
})
</script>
