<template>
  <section class="project-list-view">
    <div class="page-header">
      <div>
        <h1 class="page-title">工作台</h1>
      </div>
    </div>

    <div class="project-toolbar">
      <div class="project-toolbar-actions">
        <el-input
          v-model="projectKeyword"
          :prefix-icon="Search"
          clearable
          placeholder="搜索项目名称、ID 或描述"
          @keyup.enter="reloadProjects"
        />
        <el-button type="primary" :icon="Search" @click="reloadProjects">搜索</el-button>
        <el-button :icon="Refresh" @click="reloadProjects">刷新</el-button>
        <el-button type="primary" :icon="Plus" @click="openCreateDialog">新建项目</el-button>
      </div>
    </div>

    <div class="project-list" v-loading="projectStore.loading">
      <article
        v-for="project in projectStore.paginatedProjects"

        <span class="project-row-icon">

          <el-button type="primary" :icon="Edit" @click.stop="openEditDialog(project)">编辑</el-button>
          <el-button type="danger" plain :icon="Delete" @click.stop="removeProject(project)">删除</el-button>

      </template>

<script setup>
import { computed, onMounted, reactive, watch } from 'vue'
import { useRoute, useRouter } from 'vue-router'
import { ElMessage, ElMessageBox } from 'element-plus'
import {
  Clock,
  Delete,
  Edit,
  FolderOpened,
  Plus,
  Refresh,
  Search,
} from '@element-plus/icons-vue'

import { isProtectedProjectId, useProjectStore } from '@/stores/projects'
import { formatDate } from '@/utils/format'

const router = useRouter()
const route = useRoute()
const projectStore = useProjectStore()

const projectKeyword = computed({
  get: () => projectStore.keyword,
  set: (value) => projectStore.setKeyword(value),
})

const projectDialog = reactive({
  visible: false,
  loading: false,
  id: '',
  name: '',
  description: '',
})

function resetDialog(project = null) {
  projectDialog.id = project?.id || ''
  projectDialog.name = project?.name || ''
  projectDialog.description = project?.description || ''
}

function openCreateDialog() {
  resetDialog()
  projectDialog.visible = true
}

function openCreateDialogFromRoute() {
  if (String(route.query.create || '') !== '1') {
    return
  }

  openCreateDialog()
  const query = { ...route.query }
  delete query.create
  router.replace({
    path: '/projects',
    query,
  })
}

function openEditDialog(project) {
  if (isProtectedProject(project)) {
    ElMessage.warning('公共项目不允许编辑')
    return
  }

  resetDialog(project)
  projectDialog.visible = true
}

async function saveProject() {
  if (isProtectedProjectId(projectDialog.id)) {
    ElMessage.warning('公共项目不允许编辑')
    return
  }

  if (!projectDialog.name) {
    ElMessage.warning('请填写项目名称')
    return
  }

  try {
    projectDialog.loading = true
    await projectStore.saveProject({
      name: projectDialog.name,
      description: projectDialog.description,
    }, projectDialog.id)

    projectDialog.visible = false
    ElMessage.success(projectDialog.id ? '项目已更新' : '项目已创建')
  } catch (error) {
    if (!error?.response) {
      ElMessage.error(error.message || '项目保存失败')
    }
  } finally {
    projectDialog.loading = false
  }
}

async function removeProject(project) {
  if (!project?.id) {
    return
  }

  if (isProtectedProject(project)) {
    ElMessage.warning('公共项目不允许删除')
    return
  }

  const confirmed = await ElMessageBox.confirm(
    `确认删除项目「${project.name || project.id}」？关联模板和工作流也会被删除。`,
    '删除项目',
    { type: 'warning' },
  )
    .then(() => true)
    .catch(() => false)

  if (!confirmed) {
    return
  }

  await projectStore.removeProject(project.id)
  ElMessage.success('项目已删除')
}

function isProtectedProject(project) {
  return isProtectedProjectId(project?.id)
}

async function reloadProjects() {
  await projectStore.loadProjects()
}

function changeProjectPage(page) {
  projectStore.setPage(page)
}

function changeProjectPageSize(size) {
  projectStore.setPageSize(size)
}

function enterProject(project) {
  if (!project?.id) {
    return
  }

  projectStore.current = project
  router.push({
    name: 'project-detail',
    params: { projectId: project.id },
  })
}

onMounted(async () => {
  await projectStore.loadProjects()
  openCreateDialogFromRoute()
})

watch(() => route.query.create, openCreateDialogFromRoute)
</script>
