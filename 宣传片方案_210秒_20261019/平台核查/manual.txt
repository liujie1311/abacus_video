      </div>

      <div class="navbar-search">
        <el-input
          v-model="searchKeyword"
          :prefix-icon="Search"
          clearable
          placeholder="搜索文档..."
          @blur="hideSearchResultsSoon"
          @focus="showSearchResults = searchResults.length > 0 || Boolean(searchKeyword)"
        />
        <div v-if="showSearchResults && searchResults.length > 0" class="search-dropdown">
          <button
            v-for="result in searchResults"
            :key="result.id"
            type="button"
            class="search-item"
            @mousedown="selectSearchResult(result)"
          >
            <Document class="search-icon" />
            <span>{{ result.text }}</span>
          </button>
        </div>
        <div v-else-if="showSearchResults && searchKeyword" class="search-dropdown">
          <div class="search-empty">未找到相关内容</div>
        </div>
      </div>
    </header>

    <div class="doc-main">
      <aside class="doc-sidebar">
        <nav class="sidebar-nav">
          <div class="nav-section">
            <div class="nav-title">文档列表</div>
            <ul class="nav-list">
              <li
                v-for="item in sidebarMenu"
                :key="item.id"
                :class="['nav-item', { active: currentDocId === item.id }]"
                @click="loadDoc(item.id)"
              >
                <Document class="nav-icon" />
                <span>{{ item.title }}</span>
              </li>
            </ul>
          </div>
        </nav>

        <div v-if="tocList.length > 0" class="toc-section">
          <div class="toc-header">目录</div>
          <ul class="toc-list">
            <li
              v-for="heading in tocList"
              :key="heading.id"
              :class="['toc-item', `level-${heading.level}`, { active: activeHeading === heading.id }]"
              @click="scrollToHeading(heading.id)"
            >
              {{ heading.text }}
            </li>
          </ul>
        </div>
      </aside>

      <main ref="contentRef" v-loading="loading" class="doc-content" @scroll="handleContentScroll">
        <article v-if="docContent" class="markdown-body" v-html="renderedContent"></article>
        <el-empty v-else-if="!loading" description="请选择左侧文档查看" />
      </main>
    </div>

    <el-image-viewer
      v-if="showImageViewer"
      hide-on-click-modal
      teleported
      :initial-index="previewIndex"
      :max-scale="4"
      :min-scale="0.35"
      :scale="previewInitialScale"
      :url-list="previewImages"
      @close="showImageViewer = false"
    />
  </div>

<script setup>
import { computed, nextTick, onMounted, ref, watch } from 'vue'
import { Document, Search } from '@element-plus/icons-vue'
import { ElImageViewer } from 'element-plus'
import { marked } from 'marked'
import 'element-plus/theme-chalk/el-image-viewer.css'

import userManualMd from './使用手册/用户手册.md?raw'

const localImageModules = import.meta.glob('./使用手册/images/*', {
  eager: true,
  import: 'default',
  query: '?url',
})

const loading = ref(false)
const docContent = ref('')
const searchKeyword = ref('')
const currentDocId = ref('user-manual')
const activeHeading = ref('')
const tocList = ref([])
const contentRef = ref(null)
const searchResults = ref([])
const showSearchResults = ref(false)
const showImageViewer = ref(false)
const previewImages = ref([])
const previewIndex = ref(0)
const previewInitialScale = 0.72

const sidebarMenu = [
  { id: 'user-manual', title: '用户手册', content: userManualMd },
]

const imageUrlMap = Object.entries(localImageModules).reduce((map, [path, url]) => {
  const fileName = path.split('/').pop()
  map[fileName] = url
  map[`images/${fileName}`] = url
  return map
}, {})

const displayContent = computed(() => stripManualToc(docContent.value))
const renderedContent = computed(() => renderMarkdown(displayContent.value))
let headingRenderIds = new Set()

const markdownRenderer = new marked.Renderer()

markdownRenderer.heading = function ({ tokens, depth }) {
  const html = this.parser.parseInline(tokens)
  const text = stripHtml(html)
  const id = ensureUniqueId(slugify(text), headingRenderIds)

  return `<h${depth} id="${escapeAttr(id)}">${html}</h${depth}>`
}

markdownRenderer.image = function ({ href, title, text }) {
  const src = resolveImageUrl(href)
  const titleAttr = title ? ` title="${escapeAttr(title)}"` : ''

  return `<img src="${escapeAttr(src)}" alt="${escapeAttr(text)}"${titleAttr} loading="lazy">`
}

markdownRenderer.link = function ({ href, title, tokens }) {
  const html = this.parser.parseInline(tokens)
  const titleAttr = title ? ` title="${escapeAttr(title)}"` : ''
  const targetAttr = String(href || '').startsWith('#') ? '' : ' target="_blank" rel="noopener noreferrer"'

  return `<a href="${escapeAttr(href)}"${titleAttr}${targetAttr}>${html}</a>`
}

function stripManualToc(markdown) {
  const lines = String(markdown || '').replace(/\r\n/g, '\n').split('\n')
  const tocStartIndex = lines.findIndex((line) => /^##\s+目录\s*$/.test(line.trim()))

  if (tocStartIndex === -1) {
    return lines.join('\n')
  }

  let removeStartIndex = tocStartIndex
  while (removeStartIndex > 0 && !lines[removeStartIndex - 1].trim()) {
    removeStartIndex -= 1
  }
  if (removeStartIndex > 0 && /^-{3,}$/.test(lines[removeStartIndex - 1].trim())) {
    removeStartIndex -= 1
  }

  let removeEndIndex = tocStartIndex + 1
  while (removeEndIndex < lines.length && !/^-{3,}$/.test(lines[removeEndIndex].trim())) {
    removeEndIndex += 1
  }
  if (removeEndIndex < lines.length && /^-{3,}$/.test(lines[removeEndIndex].trim())) {
    removeEndIndex += 1
  }
  while (removeEndIndex < lines.length && !lines[removeEndIndex].trim()) {
    removeEndIndex += 1
  }

  return [
    ...lines.slice(0, removeStartIndex),
    '',
    ...lines.slice(removeEndIndex),
  ].join('\n')
}

function slugify(text) {
  return String(text || '')
    .trim()
    .replace(/<[^>]+>/g, '')
    .replace(/[\\/#?%*:|"<>.]/g, '')
    .replace(/\s+/g, '-')
    .toLowerCase()
}

function ensureUniqueId(baseId, usedIds) {
  const fallback = 'section'
  const normalized = baseId || fallback
  let id = normalized
  let index = 2

  while (usedIds.has(id)) {
    id = `${normalized}-${index}`
    index += 1
  }

  usedIds.add(id)
  return id
}

function escapeHtml(value) {
  return String(value ?? '')
    .replace(/&/g, '&amp;')
    .replace(/</g, '&lt;')
    .replace(/>/g, '&gt;')
    .replace(/"/g, '&quot;')
    .replace(/'/g, '&#39;')
}

function escapeAttr(value) {
  return escapeHtml(value).replace(/`/g, '&#96;')
}

function resolveImageUrl(path) {
  const rawPath = String(path || '').trim()
  const normalizedPath = rawPath.replace(/\\/g, '/').replace(/^\.\//, '')
  const fileName = normalizedPath.split('/').pop()

  return imageUrlMap[normalizedPath] || imageUrlMap[fileName] || rawPath
}

function stripHtml(value) {
  return String(value || '')
    .replace(/<[^>]*>/g, '')
    .replace(/&nbsp;/g, ' ')
    .replace(/&amp;/g, '&')
    .replace(/&lt;/g, '<')
    .replace(/&gt;/g, '>')
    .replace(/&quot;/g, '"')
    .replace(/&#39;/g, "'")
}

function renderMarkdown(markdown) {
  headingRenderIds = new Set()

  const html = marked.parse(String(markdown || ''), {
    async: false,
    breaks: false,
    gfm: true,
    renderer: markdownRenderer,
  })

  return typeof html === 'string' ? html : ''
}

function extractToc() {
  const headingRegex = /^(#{1,3})\s+(.+)$/gm
  const headings = []
  const usedIds = new Set()
  let match

  while ((match = headingRegex.exec(displayContent.value)) !== null) {
    const level = match[1].length
    const text = match[2].replace(/#+$/, '').trim()
    const id = ensureUniqueId(slugify(text), usedIds)
    headings.push({ level, text, id })
  }

  tocList.value = headings
  activeHeading.value = headings[0]?.id || ''
}

function scrollToHeading(id) {
  activeHeading.value = id
  const heading = contentRef.value?.querySelector(`#${CSS.escape(id)}`)

  if (heading && contentRef.value) {
    contentRef.value.scrollTo({
      top: heading.offsetTop - 20,
      behavior: 'smooth',
    })
  }
}

function handleContentScroll() {
  if (!contentRef.value || tocList.value.length === 0) {
    return
  }

  const containerRect = contentRef.value.getBoundingClientRect()
  const headings = Array.from(contentRef.value.querySelectorAll('h1, h2, h3'))
  let currentId = ''

  headings.forEach((heading) => {
    const rect = heading.getBoundingClientRect()
    if (rect.top <= containerRect.top + 100) {
      currentId = heading.id
    }
  })

  if (currentId && currentId !== activeHeading.value) {
    activeHeading.value = currentId
  }
}

function handleSearch() {
  const keyword = searchKeyword.value.trim().toLowerCase()

  if (!keyword) {
    searchResults.value = []
    showSearchResults.value = false
    return
  }

  searchResults.value = tocList.value
    .filter((item) => item.text.toLowerCase().includes(keyword))
    .slice(0, 12)
  showSearchResults.value = true
}

function selectSearchResult(result) {
  scrollToHeading(result.id)
  showSearchResults.value = false
  searchKeyword.value = ''
}

function hideSearchResultsSoon() {
  window.setTimeout(() => {
    showSearchResults.value = false
  }, 200)
}

function bindImagePreview() {
  const images = Array.from(contentRef.value?.querySelectorAll('.markdown-body img') || [])
  const urls = images.map((image) => image.src)

  images.forEach((image, index) => {
    image.style.cursor = 'zoom-in'
    image.addEventListener('click', () => {
      previewImages.value = urls
      previewIndex.value = index
      showImageViewer.value = true
    })
    image.addEventListener('error', () => {
      image.classList.add('is-missing')
      image.alt = `${image.alt || '图片'}（未找到）`
    })
  })
}

async function loadDoc(docId) {
  const doc = sidebarMenu.find((item) => item.id === docId)

  if (!doc) {
    return
  }

  currentDocId.value = docId
  loading.value = true

  try {
    docContent.value = doc.content
    await nextTick()
    extractToc()
    await nextTick()
    contentRef.value?.scrollTo({ top: 0 })
    bindImagePreview()
  } finally {
    loading.value = false
  }
}

watch(searchKeyword, handleSearch)

onMounted(() => {
  loadDoc(currentDocId.value)
})
</script>
