      incar: '',
      kpoints: '',
    },
  }

function normalizeElement(value) {
  const text = String(value || '').trim()

  if (!text) {
    return ''
  }

  return text.charAt(0).toUpperCase() + text.slice(1).toLowerCase()
}

async function setFile(file, state, type) {
  if (!file) {
    state.file = null
    state.output = ''
    state.mappings = []
    return
  }

  if (file.size > maxUploadSize) {
    ElMessage.warning('文件超过 20MB，请确认是否选择了正确的结构文件')
    return
  }

  state.file = file
  state.sourceMode = 'file'
  state.hostPath = ''
  state.output = ''
  state.mappings = []

  await prepareStructureMapping(type, state)
}

function setStructureSourceMode(state, mode) {
  if (!['file', 'path'].includes(mode) || state.sourceMode === mode) {
    return
  }

  state.sourceMode = mode
  state.output = ''
  state.mappings = []

  if (mode === 'path') {
    state.file = null
    state.inputKey += 1
  } else {
    state.hostPath = ''
  }
}

function setStructureHostPath(state, path) {
  const nextPath = String(path || '')

  if (state.hostPath === nextPath) {
    return
  }

  state.sourceMode = 'path'
  state.hostPath = nextPath
  state.output = ''
  state.mappings = []
}

function createMapping(element = '') {
  const normalized = normalizeElement(element)

  return {
    element: normalized,
    pp: '',
    ppChoice: '',
    basis: '',
    pathMode: 'preset',
    pathCandidates: [],
    ppOptions: [],
    basisOptions: [],
    mass: '',
    magmom: '',
  }
}

function removeMapping(state, index) {
  state.mappings.splice(index, 1)
}

async function detectElements(type, { silent = false } = {}) {
  const state = type === 'cif' ? cifState : poscarState
  const detector = type === 'cif' ? detectCifElements : detectPoscarElements
  const source = getStructureSource(state)

  if (!source) {
    if (!silent) {
      ElMessage.warning(state.sourceMode === 'path' ? '请先填写宿主机文件路径' : '请先选择文件')
    }
    return
  }

  try {
    state.detecting = true
    const result = await detector(source)

    if (result?.error) {
      ElMessage.warning(result.error)
    }

    const elements = Array.isArray(result?.elements) ? result.elements : []
    state.mappings = elements.map((element) => createMapping(element))

    if (elements.length) {
      try {
        await loadMappingPpPathOptions(state.mappings)
      } catch (error) {
        ElMessage.warning('已检测到元素，但默认路径加载失败，请手动填写 PP/Orb 路径')
      }
    }

    if (!silent) {
      ElMessage.success(elements.length ? `检测到 ${elements.length} 个元素，已填充默认 PP/Orb 路径` : '未检测到元素')
    }
    return
  } finally {
    state.detecting = false
  }
}

async function fillDefaultMasses(state, { silent = false } = {}) {
  if (!state.mappings.length) {
    if (!silent) {
      ElMessage.warning('请先检测元素')
    }
    return
  }

  if (!defaultMasses.value) {
    defaultMasses.value = await getDefaultMasses()
  }

  state.mappings.forEach((mapping) => {
    const element = normalizeElement(mapping.element)
    const mass = defaultMasses.value?.[element]

    if (mass !== undefined && mapping.mass === '') {
      mapping.mass = String(mass)
    }
  })

  if (!silent) {
    ElMessage.success('已填充默认质量')
  }
}

async function fillDefaultPpPaths(state, { silent = false } = {}) {
  if (!state.mappings.length) {
    if (!silent) {
      ElMessage.warning('请先检测元素')
    }
    return
  }

  const ppPathMap = await loadDefaultPpPathMap()
  let filledCount = 0
  let matchedCount = 0

  state.mappings.forEach((mapping) => {
    const element = normalizeElement(mapping.element)
    const paths = ppPathMap[element]

    if (!paths) {
      return
    }

    matchedCount += 1
    const previousPp = String(mapping.pp || '')
    const previousBasis = String(mapping.basis || '')

    applyMappingPathCandidates(mapping, paths.candidates, { preferDefault: true })

    if (mapping.pp && mapping.pp !== previousPp) {
      filledCount += 1
    }

    if (mapping.basis && mapping.basis !== previousBasis) {
      filledCount += 1
    }
  })

  if (!silent) {
    ElMessage.success(
      filledCount
        ? '已填充默认赝势/轨道路径'
        : matchedCount
          ? '当前已是默认赝势/轨道路径'
          : '未找到可填充的默认路径',
    )
  }
}

async function loadMappingPpPathOptions(mappings, { preserveExisting = true } = {}) {
  const ppPathMap = await loadDefaultPpPathMap()
  let optionCount = 0

  mappings.forEach((mapping) => {
    const element = normalizeElement(mapping.element)
    const paths = ppPathMap[element]

    if (!paths) {
      return
    }

    applyMappingPathCandidates(mapping, paths.candidates, { preserveExisting })

    optionCount += paths.candidates.reduce((count, candidate) => (
      count + candidate.ppFiles.length + candidate.orbitalFiles.length
    ), 0)
  })

  return optionCount
}

async function loadDefaultPpPathMap() {
  if (defaultPpPathMap.value && Object.keys(defaultPpPathMap.value).length) {
    return defaultPpPathMap.value
  }

  defaultPpPathMap.value = buildDefaultPpPathMap(await getDefaultPpPaths(), normalizeElement)

  return defaultPpPathMap.value
}

function uniquePaths(paths) {
  return [...new Set(paths.map((path) => String(path || '').trim()).filter(Boolean))]
}

function getMappingPpOptions(mapping) {
  const candidates = getMappingPathCandidates(mapping)

  if (candidates.length) {
    return candidates.map((candidate) => ({
      value: candidate.id,
      label: formatCandidateOptionLabel(candidate),
    }))
  }

  return uniquePaths([mapping.pp, ...(Array.isArray(mapping.ppOptions) ? mapping.ppOptions : [])])
    .map((path) => ({
      value: path,
      label: formatPathOptionLabel(path),
    }))
}

function getMappingBasisOptions(mapping) {
  return uniquePaths([
    mapping.basis,
    ...getMatchingOrbitalFiles(mapping),
  ])
}

function formatPathOptionLabel(path) {
  const parts = String(path || '').replaceAll('\\', '/').split('/').filter(Boolean)
  return parts.slice(-4).join('/') || String(path || '')
}

function applyMappingPathCandidates(mapping, candidates, { preserveExisting = false, preferDefault = false } = {}) {
  mapping.pathCandidates = Array.isArray(candidates) ? candidates : []
  mapping.ppOptions = getMappingPathCandidates(mapping).flatMap((candidate) => candidate.ppFiles)

  const pathCandidates = getMappingPathCandidates(mapping)
  const selectedCandidate = preferDefault ? pathCandidates[0] : findCandidateForMapping(mapping) || pathCandidates[0]

  if (preferDefault || !preserveExisting || !String(mapping.ppChoice || '').trim() || !findCandidateById(mapping, mapping.ppChoice)) {
    mapping.ppChoice = selectedCandidate?.id || ''
    mapping.pp = selectedCandidate?.ppFiles?.[0] || ''
    mapping.pathMode = 'preset'
  }

  updateMappingOrbitalOptions(mapping, { preserveExisting: preserveExisting && !preferDefault })
}

async function prepareStructureMapping(type, state) {
  const source = getStructureSource(state)

  if (!source) {
    ElMessage.warning(state.sourceMode === 'path' ? '请先填写宿主机文件路径' : '请先选择文件')
    return
  }

  await detectElements(type, { silent: true })

  if (!state.mappings.length) {
    ElMessage.warning('未检测到元素')
    return
  }

  await fillDefaultMasses(state, { silent: true })
  await fillDefaultPpPaths(state, { silent: true })
  ElMessage.success(`检测到 ${state.mappings.length} 个元素，已自动填充默认映射`)
}

function handleMappingPpChange(mapping, value) {
  mapping.ppChoice = value
  const selectedCandidate = findCandidateById(mapping, value)

  if (selectedCandidate) {
    mapping.pp = selectedCandidate.ppFiles[0] || ''
    mapping.pathMode = 'preset'
  } else {
    mapping.pp = value
    mapping.pathMode = 'custom'
  }

  updateMappingOrbitalOptions(mapping)
}

function handleMappingPathModeChange(mapping, mode) {
  mapping.pathMode = mode

  if (mode === 'preset') {
    const selectedCandidate = findCandidateById(mapping, mapping.ppChoice) || getMappingPathCandidates(mapping)[0]

    if (selectedCandidate) {
      mapping.ppChoice = selectedCandidate.id
      mapping.pp = selectedCandidate.ppFiles[0] || ''
      updateMappingOrbitalOptions(mapping)
    }
  }
}

function handleMappingCustomPpChange(mapping) {
  mapping.ppChoice = ''
  mapping.pathMode = 'custom'
}

function updateMappingOrbitalOptions(mapping, { preserveExisting = false } = {}) {
  const orbitalFiles = getMatchingOrbitalFiles(mapping)
  mapping.basisOptions = orbitalFiles

  if (preserveExisting && orbitalFiles.includes(mapping.basis)) {
    return
  }

  mapping.basis = orbitalFiles[0] || ''
}

function getMatchingOrbitalFiles(mapping) {
  const candidates = getMappingPathCandidates(mapping)
  const selectedCandidate = findCandidateById(mapping, mapping.ppChoice)

  if (selectedCandidate) {
    return uniquePaths(selectedCandidate.orbitalFiles)
  }

  const pp = String(mapping.pp || '').trim()

  if (!pp) {
    return uniquePaths(candidates.flatMap((candidate) => candidate.orbitalFiles))
  }

  const matchedCandidates = candidates.filter((candidate) => candidate.ppFiles.includes(pp))
  return uniquePaths(matchedCandidates.flatMap((candidate) => candidate.orbitalFiles))
}

function getMappingPathCandidates(mapping) {
  return Array.isArray(mapping.pathCandidates) ? mapping.pathCandidates : []
}

function findCandidateById(mapping, id) {
  return getMappingPathCandidates(mapping).find((candidate) => candidate.id === id)
}

function findCandidateForMapping(mapping) {
  const pp = String(mapping.pp || '').trim()
  const basis = String(mapping.basis || '').trim()

  return getMappingPathCandidates(mapping).find((candidate) => (
    (!pp || candidate.ppFiles.includes(pp)) &&
    (!basis || candidate.orbitalFiles.includes(basis))
  ))
}

function formatCandidateOptionLabel(candidate) {
  const ppSet = candidate.ppSet || 'PP'
  const orbitalSet = candidate.orbitalSet || 'Orb'
  const ppName = formatPathOptionLabel(candidate.ppFiles[0] || '')
  return `${ppSet} -> ${orbitalSet} · ${ppName}`
}

function parseOptionalNumber(value, label) {
  const text = String(value ?? '').trim()

  if (!text) {
    return undefined
  }

  const parsed = Number(text)

  if (!Number.isFinite(parsed)) {
    throw new Error(`${label} 必须是数字`)
  }

  return parsed
}

function normalizeHostFilePath(path) {
  return String(path || '').trim().replaceAll('\\', '/').replace(/\/+$/, '')
}

function getStructureSource(state) {
  if (state.sourceMode === 'path') {
    const path = normalizeHostFilePath(state.hostPath)
    return path ? { content: '', path } : null
  }

  return state.file || null
}

function getStructureSourceLabel(state) {
  if (state.sourceMode === 'path') {
    return '宿主机文件路径'
  }

  return '文件'
}

function getFileNameFromPath(path, fallback) {
  const fileName = normalizeHostFilePath(path).split('/').filter(Boolean).pop()
  return fileName || fallback
}

async function getStructureUploadFile(state, type) {
  if (state.sourceMode !== 'path') {
    return state.file
  }

  const path = normalizeHostFilePath(state.hostPath)

  if (!path) {
    throw new Error('请先填写宿主机文件路径')
  }

  const file = await readServerFile(path)
  const content = file?.content

  if (typeof content !== 'string') {
    throw new Error('宿主机文件读取失败，未返回文本内容')
  }

  return new File([content], getFileNameFromPath(file?.path || path, type === 'cif' ? 'structure.cif' : 'POSCAR'), {
    type: 'text/plain;charset=utf-8',
  })
}

async function buildStructureFormData(state, type) {
  if (!getStructureSource(state)) {
    throw new Error(`请先选择${getStructureSourceLabel(state)}`)
  }

  if (!state.mappings.length) {
    throw new Error('请配置元素映射')
  }

  const sourceFile = await getStructureUploadFile(state, type)

  const pp = {}
  const basis = {}
  const masses = {}
  const magmom = {}
  const elements = []
  const seenElements = new Set()

  // 后端结构转换接口要求映射字段以 JSON 字符串放在 multipart 表单里。
  state.mappings.forEach((mapping) => {
    const element = normalizeElement(mapping.element)

    if (!element) {
      throw new Error('元素符号不能为空')
    }

    if (seenElements.has(element)) {
      throw new Error(`元素 ${element} 配置重复`)
    }

    seenElements.add(element)
    elements.push(element)

    const ppFile = String(mapping.pp || '').trim()
    const basisFile = String(mapping.basis || '').trim()

    if (!ppFile || !basisFile) {
      const missingPaths = [!ppFile ? 'PP' : '', !basisFile ? 'Orb' : ''].filter(Boolean).join(' 和 ')
      throw new Error(`元素 ${element} 请先选择或填写 ${missingPaths} 路径`)
    }

    pp[element] = ppFile
    basis[element] = basisFile

    const mass = parseOptionalNumber(mapping.mass, `${element} 质量`)
    const magneticMoment = parseOptionalNumber(mapping.magmom, `${element} 磁矩`)

    if (mass !== undefined) {
      masses[element] = mass
    }

    if (magneticMoment !== undefined) {
      magmom[element] = magneticMoment
    }
  })

  const formData = new FormData()
  formData.append('file', sourceFile)
  formData.append('elements', JSON.stringify(elements))
  formData.append('pp', JSON.stringify(pp))
  formData.append('basis', JSON.stringify(basis))

  if (Object.keys(masses).length) {
    formData.append('masses', JSON.stringify(masses))
  }

  if (Object.keys(magmom).length) {
    formData.append('magmom', JSON.stringify(magmom))
  }

  return formData
}

async function convertStructure(type) {
  const state = type === 'cif' ? cifState : poscarState
  const converter = type === 'cif' ? convertCifToStru : convertPoscarToStru

  try {
    state.converting = true
    state.output = await converter(await buildStructureFormData(state, type))
    ElMessage.success('转换完成')
  } catch (error) {
    if (error?.message && !error?.response) {
      ElMessage.warning(error.message)
    }
  } finally {
    state.converting = false
  }
}

function clearStructureState(state) {

  state.hostPath = ''

  state.output = ''
}

async function setVaspDirectory(event) {
  const files = Array.from(event.target.files || [])

  resetVaspDirectoryState({ sourceMode: 'local' })
  clearStructureState(poscarState)

  if (!files.length) {
    return
  }

  vaspState.directoryName = getVaspDirectoryName(files)
  vaspState.poscarFile = findVaspDirectoryFile(files, 'POSCAR')
  vaspState.incarFile = findVaspDirectoryFile(files, 'INCAR')
  vaspState.kpointsFile = findVaspDirectoryFile(files, 'KPOINTS')
  clearVaspIncarStructureSelection()
  selectFirstVaspResultTab()

  if (!vaspState.poscarFile && !vaspState.incarFile && !vaspState.kpointsFile) {
    ElMessage.warning('所选目录中未找到 POSCAR、INCAR 或 KPOINTS 文件')
    return
  }

  await prepareVaspPoscarConfiguration()
}

function setVaspSourceMode(mode) {
  if (!['local', 'path'].includes(mode)) {
    return
  }

  resetVaspDirectoryState({ sourceMode: mode })
  clearStructureState(poscarState)
}

function setVaspHostPath(path) {

  resetVaspDirectoryState({

    hostPath: nextPath,

async function readVaspHostDirectory() {
  const directoryPath = normalizeHostDirectoryPath(vaspState.hostPath)

  if (!directoryPath) {
    ElMessage.warning('请先填写宿主机 VASP 计算目录')
    return
  }

  resetVaspDirectoryState({
    sourceMode: 'path',
    hostPath: directoryPath,
    directoryName: directoryPath,
  })
  clearStructureState(poscarState)

  try {
    vaspState.reading = true
    const files = await Promise.all([
      readVaspHostFile(directoryPath, 'POSCAR'),
      readVaspHostFile(directoryPath, 'INCAR'),
      readVaspHostFile(directoryPath, 'KPOINTS'),
    ])

    vaspState.poscarFile = files[0]
    vaspState.incarFile = files[1]
    vaspState.kpointsFile = files[2]
    clearVaspIncarStructureSelection()
    selectFirstVaspResultTab()

    if (!hasVaspSourceFiles.value) {
      ElMessage.warning('该目录中未找到 POSCAR、INCAR 或 KPOINTS 文件')
      return
    }

    await prepareVaspPoscarConfiguration()
  } finally {
    vaspState.reading = false
  }
}

async function readVaspHostFile(directoryPath, filename) {
  try {
    const file = await readServerFile(joinHostDirectoryPath(directoryPath, filename), { silent: true })
    const content = file?.content

    if (typeof content !== 'string') {
      return null
    }

    return new File([content], filename, { type: 'text/plain;charset=utf-8' })
  } catch {
    return null
  }
}

function normalizeHostDirectoryPath(path) {
  return String(path || '').trim().replaceAll('\\', '/').replace(/\/+$/, '')
}

function joinHostDirectoryPath(directoryPath, filename) {
  return `${normalizeHostDirectoryPath(directoryPath)}/${filename}`
}

async function convertVaspDirectory() {
  if (!hasVaspSourceFiles.value) {
    ElMessage.warning('请先读取包含 POSCAR、INCAR 或 KPOINTS 的 VASP 计算目录')
    return
  }

  const conversions = [
    ['poscar', vaspState.poscarFile, convertVaspPoscar],
    ['incar', vaspState.incarFile, convertVaspIncar],
    ['kpoints', vaspState.kpointsFile, convertVaspKpoints],
  ]

  vaspState.converting = true
  vaspState.struOutput = ''
  vaspState.inputOutput = ''
  vaspState.kptOutput = ''
  vaspState.incarWarnings = []
  vaspState.magmom = null

  conversions.forEach(([key, file]) => {
    vaspState.status[key] = file ? 'converting' : 'missing'
    vaspState.errors[key] = ''
  })

  try {
    await Promise.all(conversions
      .filter(([, file]) => file)
      .map(([, , converter]) => converter()))

    const completedCount = Object.values(vaspState.status).filter((status) => status === 'success').length
    if (completedCount) {
      ElMessage.success(`已生成 ${completedCount} 个 ABACUS 输入文件`)
    }
  } finally {
    vaspState.converting = false
  }
}

async function convertVaspPoscar() {
  if (!vaspState.poscarFile) {
    return
  }

  try {
    if (!poscarState.mappings.length) {
      throw new Error('请先完成 POSCAR 的元素映射配置')
    }

    vaspState.struOutput = await convertPoscarToStru(await buildStructureFormData(poscarState, 'poscar'))
    vaspState.status.poscar = 'success'
  } catch (error) {
    vaspState.status.poscar = 'error'
    vaspState.errors.poscar = getVaspConversionError(error)
  }
}

async function convertVaspIncar() {
  if (!vaspState.incarFile) {
    return
  }

  try {
    const formData = new FormData()
    formData.append('incar', vaspState.incarFile)

    if (vaspState.incarStructureSource === 'custom' && vaspState.incarStructureFile) {
      formData.append('stru', vaspState.incarStructureFile)
    }

    const result = await convertIncarToInput(formData)
    vaspState.inputOutput = typeof result === 'string' ? result : String(result?.input || '')
    vaspState.incarWarnings = Array.isArray(result?.warnings) ? result.warnings : []
    vaspState.magmom = result?.magmom ?? null

    if (!vaspState.inputOutput.trim()) {
      throw new Error('INCAR 转换接口未返回 INPUT 内容')
    }

    vaspState.status.incar = 'success'
  } catch (error) {
    vaspState.status.incar = 'error'
    vaspState.errors.incar = getVaspConversionError(error)
  }
}

async function convertVaspKpoints() {
  if (!vaspState.kpointsFile) {
    return
  }

  try {
    const formData = new FormData()
    formData.append('file', vaspState.kpointsFile)

    const result = await convertKpointsToKpt(formData)
    vaspState.kptOutput = typeof result === 'string' ? result : String(result?.kpt || '')

    if (!vaspState.kptOutput.trim()) {
      throw new Error('KPOINTS 转换接口未返回 KPT 内容')
    }

    vaspState.status.kpoints = 'success'
  } catch (error) {
    vaspState.status.kpoints = 'error'
    vaspState.errors.kpoints = getVaspConversionError(error)
  }
}

function getVaspConversionError(error) {
  const responseData = error?.response?.data
  const message = typeof responseData === 'string'
    ? responseData
    : responseData?.error || responseData?.message || error?.message

  return String(message || '转换失败，请检查源文件内容').trim()
}

function getVaspFileStatus(key, source, target, file) {
  const status = vaspState.status[key]
  const error = vaspState.errors[key]
  const configAction = getVaspFileConfigAction(key)

  if (!file) {
    return { key, source, target, file, description: '', status, tagType: 'info', message: '缺少文件', configAction }
  }

  if (status === 'converting') {
    return { key, source, target, file, description: getVaspFileDescription(key), status, tagType: 'warning', message: '转换中', configAction }
  }

  if (status === 'success') {
    return { key, source, target, file, description: getVaspFileDescription(key), status, tagType: 'success', message: '已生成', configAction }
  }

  if (status === 'error') {
    return { key, source, target, file, description: getVaspFileDescription(key), status, tagType: 'danger', message: error || '转换失败', configAction }
  }

  return { key, source, target, file, description: getVaspFileDescription(key), status, tagType: 'info', message: '等待转换', configAction }
}

function getVaspFileConfigAction(key) {
  const actions = {
    poscar: '配置 STRU',
  }

  return actions[key] || ''
}

function getVaspFileDescription(key) {
  const descriptions = {
    poscar: '结构信息',
    incar: '计算参数',
    kpoints: 'K 点设置',
  }

  return descriptions[key] || ''
}

async function generatePyatb() {
  if (!pyatbState.properties.length) {
    ElMessage.warning('请至少选择一个计算属性')
    return
  }

  if (pyatbState.sourceMode === 'path' && !pyatbState.abacusPath) {
    ElMessage.warning('请填写 ABACUS 目录')
    return
  }

  if (pyatbState.sourceMode === 'task' && !pyatbState.taskId) {
    ElMessage.warning('请填写 ABACUS Task ID')
    return
  }

  try {
    pyatbState.loading = true
    const options = collectPyatbOptions(pyatbState.options)
    const sourcePayload = pyatbState.sourceMode === 'path'
      ? { abacus_path: pyatbState.abacusPath }
      : { abacus_task_id: pyatbState.taskId }

    pyatbState.result = await generatePyatbInput({
      ...sourcePayload,
      properties: pyatbState.properties,
      options: Object.keys(options).length ? options : undefined,
    })
    ElMessage.success('PYATB Input 已生成')
  } finally {
    pyatbState.loading = false
  }
}

function downloadText(text, filename) {
  const blob = new Blob([text], {
    type: 'text/plain;charset=utf-8',
  })
  const url = URL.createObjectURL(blob)
  const link = document.createElement('a')
  link.href = url
  link.download = filename
