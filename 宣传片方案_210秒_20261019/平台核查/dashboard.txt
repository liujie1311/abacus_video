<script setup>
import { computed, nextTick, onBeforeUnmount, onMounted, ref, watch } from 'vue'
import { useRouter } from 'vue-router'
import * as echarts from 'echarts/core'
import { PieChart } from 'echarts/charts'
import { CanvasRenderer } from 'echarts/renderers'
import {
  ArrowRight,
  CircleCheckFilled,
  Clock,
  Folder,
  Loading,
  Plus,
  WarningFilled,
} from '@element-plus/icons-vue'

import abacusLogo from '@/assets/logos/abacus-logo-c.png'
import lammpsLogo from '@/assets/logos/lammps-logo.png'
import homeHeaderBg from '@/assets/logos/home-header-bg-wide.png'
import databaseIcon from '@/assets/logos/home-database.svg'

import structureIcon from '@/assets/logos/structure-modeling.svg'
import templateIcon from '@/assets/logos/home-template.svg'
import toolboxIcon from '@/assets/logos/home-toolbox.svg'
import pyatbLogo from '@/assets/logos/pyatb-display-transparent.png'
import tb2jLogo from '@/assets/logos/tb2j-wordmark.svg'
import { navItems } from '@/config/navigation'
import { useProjectStore } from '@/stores/projects'
import { useSystemStore } from '@/stores/system'
import { useWorkflowStore } from '@/stores/workflows'
import { formatDate, workflowStatusText } from '@/utils/format'
import AiAssistantPanel from './components/AiAssistantPanel.vue'
import EngineBrandCarousel from './components/EngineBrandCarousel.vue'

echarts.use([CanvasRenderer, PieChart])

const router = useRouter()
const projectStore = useProjectStore()
const systemStore = useSystemStore()
const workflowStore = useWorkflowStore()
const propertyMenuOpen = ref(false)
const showAiAssistant = false
const workflowOverviewChartRef = ref(null)
let workflowOverviewChart = null
let workflowOverviewResizeObserver = null

const workflowOverviewColors = {
  total: '#5368ff',
  running: '#5368ff',
  completed: '#63c891',
  failed: '#df4450',
  stopped: '#9aa3b8',
}

const visibleWorkflows = computed(() => workflowStore.recentWorkflows.slice(0, 10))
const visibleProjects = computed(() => projectStore.projects.slice(0, 5))
const propertyEntries = computed(() => navItems.find((item) => item.path === '/properties')?.children || [])
const workflowStatusSummary = computed(() => {
  const summary = systemStore.workflowSummary || {}

  if (Object.keys(summary).length) {
    return summary
  }

  return workflowStore.workflows.reduce((result, workflow) => {
    const status = normalizeStatus(workflow.status)
    result[status] = (result[status] || 0) + 1
    return result
  }, {})
})
const workflowOverviewTotal = computed(() => {
  const summaryTotal = Object.values(workflowStatusSummary.value || {}).reduce((total, value) => total + Number(value || 0), 0)
  return Math.max(
    Number(systemStore.workflowTotal || 0),
    summaryTotal,
    Number(workflowStore.pagination.total || 0),
    workflowStore.workflows.length || 0,
  )
})
const workflowOverviewStats = computed(() => {
  const completed = getSummaryCount(['completed', 'success'])
  const failed = getSummaryCount(['failed'])
  const stopped = getSummaryCount(['stopped', 'cancelled', 'paused'])
  const running = getSummaryCount(['running'])

  return [
    { key: 'total', label: '总任务', value: workflowOverviewTotal.value },
    { key: 'running', label: '运行中', value: running },
    { key: 'completed', label: '已完成', value: completed },
    { key: 'failed', label: '失败', value: failed },
    { key: 'stopped', label: '已停止', value: stopped },
  ]
})
const workflowRingLegend = computed(() => (
  workflowOverviewStats.value
    .filter((item) => item.key !== 'total')
    .map((item) => ({
      ...item,
      percent: getPercent(item.value, workflowOverviewTotal.value),
    }))
))
const workflowRingChartData = computed(() => {
  if (!workflowOverviewTotal.value) {
    return [
      {
        name: '暂无任务',
        value: 1,
        itemStyle: {
          color: '#9aa3b8',
        },
      },
    ]
  }

  const knownTotal = workflowRingLegend.value.reduce((total, item) => total + Number(item.value || 0), 0)
  const remainder = Math.max(Number(workflowOverviewTotal.value || 0) - knownTotal, 0)
  const segments = workflowRingLegend.value
    .filter((item) => Number(item.value || 0) > 0)
    .map((item) => ({
      name: item.label,
      value: item.value,
      itemStyle: {
        color: workflowOverviewColors[item.key],
      },
    }))

  if (remainder) {
    segments.push({
      name: '未统计',
      value: remainder,
      itemStyle: {
        color: '#e9edf5',
      },
    })
  }

  return segments
})

const workflowRingOption = computed(() => ({
  animation: false,
  series: [
    {
      type: 'pie',
      radius: ['58%', '88%'],
      center: ['50%', '50%'],
      data: workflowRingChartData.value,
      silent: true,
      label: {
        show: false,
      },
      labelLine: {
        show: false,
      },
      emphasis: {
        disabled: true,
      },
    },
  ],
}))

async function renderWorkflowOverviewChart() {
  await nextTick()

  if (!workflowOverviewChartRef.value) {
    return
  }

  if (!workflowOverviewChart) {
    workflowOverviewChart = echarts.init(workflowOverviewChartRef.value, null, {
      renderer: 'canvas',
    })
    window.addEventListener('resize', resizeWorkflowOverviewChart)

    if (typeof ResizeObserver !== 'undefined') {
      workflowOverviewResizeObserver = new ResizeObserver(resizeWorkflowOverviewChart)
      workflowOverviewResizeObserver.observe(workflowOverviewChartRef.value)
    }
  }

  workflowOverviewChart.setOption(workflowRingOption.value, {
    notMerge: true,
    lazyUpdate: true,
  })
  resizeWorkflowOverviewChart()
}

function resizeWorkflowOverviewChart() {
  if (!workflowOverviewChart || !workflowOverviewChartRef.value) {
    return
  }

  workflowOverviewChart.resize()
}

function disposeWorkflowOverviewChart() {
  workflowOverviewResizeObserver?.disconnect()
  workflowOverviewResizeObserver = null
  window.removeEventListener('resize', resizeWorkflowOverviewChart)
  workflowOverviewChart?.dispose()
  workflowOverviewChart = null
}

watch(workflowRingOption, renderWorkflowOverviewChart, {
  deep: true,
  flush: 'post',
})

onBeforeUnmount(() => {
  disposeWorkflowOverviewChart()
})

const heroBrandCarouselTransition = 500

const quickEntries = [
  {
    title: '模板库',
    description: '常用计算模板与流程',
    path: '/templates',
    image: templateIcon,
  },
  {
    title: '结构建模',
    description: '构建与优化晶体结构',

    image: structureIcon,
  },
  {

    title: '工具箱',

    image: toolboxIcon,
  },
  {
    title: '材料数据库',
    description: '材料数据与性质查询',
    path: '',
    image: databaseIcon,
    disabled: true,
    badge: '暂未开放',
  },
]

const workflowStatusIcons = {
  completed: CircleCheckFilled,
  failed: WarningFilled,
  pending: Clock,
  running: Loading,
  success: CircleCheckFilled,
}

async function refreshAll() {
  await Promise.all([systemStore.refresh(), loadDashboardWorkflows(), projectStore.loadProjects()])
}

function openRoute(path) {
  router.push(path)
}

function openQuickEntry(entry) {
  if (!entry?.path || entry.disabled) {
    if (entry?.type === 'properties') {
      propertyMenuOpen.value = !propertyMenuOpen.value
    }

    return
  }

  propertyMenuOpen.value = false
  openRoute(entry.path)
}

function openPropertyEntry(path) {
  propertyMenuOpen.value = false
  openRoute(path)
}

function openWorkflow(workflowId) {
  if (!workflowId) return
  router.push(`/workflows/${encodeURIComponent(workflowId)}`)
}

function openProject(projectId) {
  if (!projectId) return
  router.push({ name: 'project-detail', params: { projectId } })
}

function normalizeStatus(status) {
  return String(status || 'unknown').trim().toLowerCase().replace(/[^a-z0-9_-]+/g, '-')
}

function getSummaryCount(keys) {
  return keys.reduce((total, key) => total + Number(workflowStatusSummary.value?.[key] || 0), 0)
}

function getPercent(value, total) {
  if (!total) return 0
  return Math.round((Number(value || 0) / total) * 1000) / 10
}

function getWorkflowStatusIcon(status) {
  return workflowStatusIcons[normalizeStatus(status)] || Clock
}

function getWorkflowUpdatedText(workflow) {
  return formatDate(workflow?.updated_at || workflow?.created_at || '')
}

function getProjectUpdatedText(project) {
  const value = project?.updated_at || project?.created_at

  if (!value) {
    return '最近更新'
  }

  return formatDate(value)
}

function loadDashboardWorkflows() {
  workflowStore.resetFilters()
  return workflowStore.loadWorkflows({ limit: 10, offset: 0 })
}

onMounted(async () => {
  await refreshAll()
  await renderWorkflowOverviewChart()
})
</script>
