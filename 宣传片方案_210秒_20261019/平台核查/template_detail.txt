                      :class="`type-${normalizeStepType(step.type)}`"

                      <strong>{{ step.id }}</strong>
                      <small>{{ getWorkflowTypeLabel(normalizeStepType(step.type)) }}</small>
                      <span v-if="step.dependencies !== '-'" class="workflow-node-dependency">
                        depends on {{ step.dependencies }}
                      </span>
                      <span v-if="step.workdir !== '-'" class="workflow-node-workdir">
                        workdir {{ step.workdir }}
                      </span>
                    </button>
                    <div v-if="index < currentWorkflowSteps.length - 1" class="workflow-graph-arrow" aria-hidden="true">
                      <span></span>
                    </div>
                  </template>
                </div>
              </div>
            </div>
            <div v-else class="empty-block compact">暂无步骤定义</div>
          </section>

          <section class="detail-section">

          <div v-else class="empty-block compact">{{ section.emptyText }}</div>

<script setup>
import { computed, reactive, ref, watch } from 'vue'
import { useRoute } from 'vue-router'
import { ElMessage } from 'element-plus'
import {
  Clock,
  Folder,
  List,
} from '@element-plus/icons-vue'

import { useTemplateStore } from '@/stores/templates'
import { translateKnownText } from '@/i18n'
import { useAppStore } from '@/stores/app'
import { formatDate } from '@/utils/format'
import abacusCoverLogo from '@/assets/logos/abacus-logo.svg'
import abacusTb2jCover from '@/assets/logos/abacus+TB2J.jpeg'
import lammpsLogo from '@/assets/logos/lammps-logo.png'
import pyatbLogoCropped from '@/assets/logos/pyatb-logo-cropped.jpg'
import scriptCoverLogo from '@/assets/logos/script-display-fill.png'
import mixedWorkflowCover from '@/assets/logos/mixed-logo.png'
import DataPreview from './components/DataPreview.vue'

const route = useRoute()
const appStore = useAppStore()
const templateStore = useTemplateStore()

const stepDetailDrawer = reactive({
  visible: false,
})
const selectedStep = ref(null)

const workflowTypeLabels = {
  abacus: 'ABACUS',
  lammps: 'LAMMPS',
  tb2j: 'TB2J',
  pyatb: 'PYATB',
  script: 'Script',
}

const templateCoverImages = {
  abacus: {
    src: abacusCoverLogo,
    alt: 'ABACUS 模板封面',
  },
  pyatb: {
    src: pyatbLogoCropped,
    alt: 'PYATB 模板封面',
  },
  lammps: {
    src: lammpsLogo,
    alt: 'LAMMPS 模板封面',
  },
  tb2j: {
    src: scriptCoverLogo,
    alt: 'TB2J 模板封面',
  },
  script: {
    src: scriptCoverLogo,
    alt: 'Script 模板封面',
  },
  mixed: {
    src: mixedWorkflowCover,
    alt: '组合工作流模板封面',
  },
  mixedAbacusTb2j: {
    src: abacusTb2jCover,
    alt: 'ABACUS + TB2J 模板封面',
  },
}

const templateId = computed(() => String(route.params.templateId || ''))
const template = computed(() => (
  String(templateStore.current?.id || '') === templateId.value ? templateStore.current : null
))
const currentWorkflowSteps = computed(() => {
  const steps = template.value?.definition?.steps

  if (!Array.isArray(steps)) {
    return []
  }

  return steps.map((step) => ({
    id: step.id || step.step_id || '-',
    type: step.type || '-',
    description: step.description || '',
    workdir: formatStepWorkdir(step),
    dependencies: formatDependencies(step),
    raw: step,
  }))
})
const selectedStepTitle = computed(() => (selectedStep.value?.id ? `步骤详情：${selectedStep.value.id}` : '步骤详情'))
const selectedStepSections = computed(() => (selectedStep.value ? buildStepDetailSections(selectedStep.value) : []))

async function loadTemplate() {
  if (!templateId.value) {
    return
  }

  try {
    await templateStore.selectTemplate(templateId.value)
  } catch (error) {
    if (!error?.response) {
      ElMessage.error(error.message || '模板加载失败')
    }
  }
}

function getTemplateTypeLabel(definition) {
  const types = resolveWorkflowTypes(definition)

  if (!types.length) {
    return '未知'
  }

  if (types.length === 1) {
    return getWorkflowTypeLabel(types[0])
  }

  return types.map((type) => getWorkflowTypeLabel(type)).join(' + ')
}

function getTemplateTypeKey(definition) {
  const types = resolveWorkflowTypes(definition)
  return types.length > 1 ? 'mixed' : types[0] || 'abacus'
}

function getTemplateTags(sourceTemplate) {
  return Array.isArray(sourceTemplate?.tags) ? sourceTemplate.tags : []
}

function translateKnown(value) {
  return translateKnownText(value, appStore.language)
}

function getTemplateCover(sourceTemplate) {
  const key = getTemplateCoverKey(sourceTemplate?.definition)

  return templateCoverImages[key] || templateCoverImages.abacus
}

function getTemplateCoverKey(definition) {
  const types = resolveWorkflowTypes(definition)

  if (types.length === 2 && types.includes('abacus') && types.includes('tb2j')) {
    return 'mixedAbacusTb2j'
  }

  return getTemplateTypeKey(definition)
}

function getTemplateParameters(sourceTemplate) {
  return isPlainRecord(sourceTemplate?.parameters) ? sourceTemplate.parameters : {}
}

function getWorkflowTypeLabel(type) {
  return workflowTypeLabels[type] || '未知'
}

function openStepDetailDrawer(step) {
  selectedStep.value = step
  stepDetailDrawer.visible = true
}

function getStepDescription(step) {
  const description = String(step?.description || step?.raw?.description || '').trim()

  if (description) {
    return description
  }

  return normalizeStepType(step?.type) === 'script' ? 'Run script commands' : 'Prepare input files'
}

function getStepConfig(step) {
  return isPlainRecord(step?.raw?.config) ? step.raw.config : {}
}

function buildStepDetailSections(step) {
  const type = normalizeStepType(step.type)
  const config = getStepConfig(step)
  const sections = []

  if (type === 'abacus') {
    sections.push({
      key: 'input',
      title: 'INPUT 参数',
      description: 'ABACUS 计算输入参数。',
      content: formatInputContent(config.INPUT),
      emptyText: '暂无 INPUT 参数',
    })
    sections.push({
      key: 'kpt',
      title: 'KPT 配置',
      description: 'K 点采样设置。',
      content: formatConfigContent(config.KPT),
      emptyText: '暂无 KPT 配置',
    })
    sections.push({
      key: 'stru',
      title: 'STRU 内容',
      description: '结构文件内容。',
      content: formatConfigContent(config.STRU),
      emptyText: '暂无 STRU 内容',
    })
  } else if (type === 'lammps') {
    sections.push({
      key: 'input',
      title: '参数配置',
      description: 'LAMMPS INPUT 内容。',
      content: formatConfigContent(config.INPUT),
      emptyText: '暂无 INPUT 内容',
    })
    sections.push({
      key: 'stru',
      title: 'STRU 内容',
      description: 'LAMMPS data 文件内容。',
      content: formatConfigContent(config.STRU ?? config.stru),
      emptyText: '暂无 STRU 内容',
    })
  } else if (type === 'tb2j') {
    sections.push({
      key: 'parameter',
      title: 'TB2J 参数',
      description: 'TB2J 磁交换参数配置。',
      content: formatConfigContent(config.parameter),
      emptyText: '暂无 TB2J 参数',
    })
  } else if (type === 'pyatb') {
    sections.push({
      key: 'input',
      title: 'PYATB Input',
      description: 'PYATB 后处理输入内容。',
      content: formatConfigContent(config.Input ?? config.INPUT),
      emptyText: '暂无 PYATB Input 内容',
    })
    sections.push({
      key: 'stru',
      title: 'STRU 内容',
      description: '该步骤关联的结构文件内容。',
      content: formatConfigContent(config.STRU ?? config.stru),
      emptyText: '暂无 STRU 内容',
    })
  } else if (type === 'script') {
    sections.push({
      key: 'script',
      title: '脚本内容',
      description: '该步骤执行的脚本。',
      content: formatConfigContent(config.script),
      badge: config.interpreter || 'script',
      emptyText: '暂无脚本内容',
    })
  }

  const commonRows = configToRows(pickCommonStepConfig(config))

  if (commonRows.length) {
    sections.push({
      key: 'common',
      title: '执行配置',
      description: '命令、环境变量、文件和输出等附加配置。',
      content: formatRowsAsBlock(commonRows),
      emptyText: '暂无执行配置',
    })
  }

  const stepRows = configToRows(pickStepLevelConfig(step))

  if (stepRows.length) {
    sections.push({
      key: 'step',
      title: '步骤控制',
      description: 'parameters、foreach 等步骤级配置。',
      content: formatRowsAsBlock(stepRows),
      emptyText: '暂无步骤控制配置',
    })
  }

  const extraRows = configToRows(getExtraStepConfig(config, type))

  if (extraRows.length) {
    sections.push({
      key: 'extra',
      title: '其他配置',
      description: '模板中额外声明的步骤配置。',
      content: formatRowsAsBlock(extraRows),
      emptyText: '暂无其他配置',
    })
  }

  return sections
}

function pickCommonStepConfig(config) {
  if (!isPlainRecord(config)) {
    return {}
  }

  return Object.fromEntries(
    ['command', 'env', 'files', 'outputs', 'dry_run', 'input_mode'].filter((key) => config[key] !== undefined).map((key) => [key, config[key]]),
  )
}

function pickStepLevelConfig(step) {
  if (!isPlainRecord(step)) {
    return {}
  }

  return Object.fromEntries(
    ['parameters', 'foreach'].filter((key) => step[key] !== undefined).map((key) => [key, step[key]]),
  )
}

function getExtraStepConfig(config, type) {
  if (!isPlainRecord(config)) {
    return {}
  }

  const knownKeys = getKnownStepConfigKeys(type)

  return Object.fromEntries(
    Object.entries(config).filter(([key]) => !knownKeys.has(key)),
  )
}

function getKnownStepConfigKeys(type) {
  const sharedKeys = ['command', 'env', 'files', 'outputs', 'workdir', 'workDir', 'work_dir', 'working_directory']
  const typeKeys = {
    abacus: ['INPUT', 'KPT', 'STRU', 'dry_run', 'input_mode'],
    lammps: ['INPUT', 'STRU', 'stru'],
    tb2j: ['parameter'],
    pyatb: ['Input', 'INPUT', 'STRU', 'stru'],
    script: ['interpreter', 'script'],
  }

  return new Set([...sharedKeys, ...(typeKeys[normalizeStepType(type)] || [])])
}

function configToRows(value) {
  if (!isPlainRecord(value)) {
    return []
  }

  return Object.entries(value).map(([key, item]) => ({
    key,
    value: formatDisplayValue(item),
  }))
}

function formatInputContent(value) {
  if (isPlainRecord(value)) {
    const rows = configToRows(value)

    if (!rows.length) {
      return ''
    }

    const maxKeyLength = Math.max(...rows.map((row) => row.key.length))
    return ['INPUT_PARAMETERS', ...rows.map((row) => `${row.key}${' '.repeat(maxKeyLength - row.key.length + 2)}${row.value}`)].join('\n')
  }

  if (typeof value === 'string') {
    return value.trim()
  }

  return formatConfigContent(value)
}

function formatRowsAsBlock(rows = []) {
  if (!rows.length) {
    return ''
  }

  const maxKeyLength = Math.max(...rows.map((row) => row.key.length))
  return rows.map((row) => `${row.key}${' '.repeat(maxKeyLength - row.key.length + 2)}${row.value}`).join('\n')
}

function formatConfigContent(value) {
  if (value === undefined || value === null) {
    return ''
  }

  if (typeof value === 'string') {
    return value.trim()
  }

  if (Array.isArray(value)) {
    return value.map((item) => formatDisplayValue(item)).join('\n')
  }

  if (isPlainRecord(value)) {
    const struText = formatStruConfig(value)

    if (struText) {
      return struText
    }

    return configToRows(value)
      .map((row) => `${row.key}: ${row.value}`)
      .join('\n')
  }

  return String(value)
}

function formatStruConfig(value) {
  if (!isPlainRecord(value)) {
    return ''
  }

  const atomSpecies = Array.isArray(value.atom_species) ? value.atom_species : []
  const atomPositions = Array.isArray(value.atom_positions) ? value.atom_positions : []
  const latticeVectors = Array.isArray(value.lattice_vectors)
    ? value.lattice_vectors
    : Array.isArray(value.lattice_vector)
      ? value.lattice_vector
      : []

  if (!atomSpecies.length && !atomPositions.length && !latticeVectors.length && value.lattice_constant === undefined) {
    return ''
  }

  const lines = []

  if (atomSpecies.length) {
    lines.push('ATOMIC_SPECIES')
    atomSpecies.forEach((species) => {
      const element = species.element || species.type || species.name || ''
      const mass = species.mass ?? species.atomic_mass ?? ''
      const pp = species.pseudopotential || species.pp || species.pp_file || species.pseudo || ''
      lines.push([element, mass, pp].map((item) => formatInlineValue(item)).filter(Boolean).join(' '))
    })
    lines.push('')

    const orbitalLines = atomSpecies
      .map((species) => species.orbital || species.basis || species.numerical_orbital)
      .filter(Boolean)

    if (orbitalLines.length) {
      lines.push('NUMERICAL_ORBITAL')
      lines.push(...orbitalLines.map((item) => formatInlineValue(item)))
      lines.push('')
    }
  }

  if (value.lattice_constant !== undefined) {
    lines.push('LATTICE_CONSTANT')
    lines.push(formatInlineValue(value.lattice_constant))
    lines.push('')
  }

  if (latticeVectors.length) {
    lines.push('LATTICE_VECTORS')
    latticeVectors.forEach((vector) => lines.push(formatInlineValue(vector)))
    lines.push('')
  }

  if (atomPositions.length) {
    lines.push('ATOMIC_POSITIONS')
    lines.push(formatCoordinateType(value.coordinate_type || value.coord_type || value.type || 'Direct'))
    atomPositions.forEach((atom) => lines.push(formatAtomPosition(atom)))
  }

  return lines.join('\n').trim()
}

function formatAtomPosition(atom) {
  if (!isPlainRecord(atom)) {
    return formatInlineValue(atom)
  }

  const element = atom.element || atom.type || atom.name || ''
  const position = atom.position || atom.positions || atom.coordinate || atom.coordinates || atom.value || []
  const tail = [atom.magmom, atom.velocity, atom.constraint].filter((item) => item !== undefined).map((item) => formatInlineValue(item))

  return [element, formatInlineValue(position), ...tail].filter(Boolean).join(' ')
}

function formatCoordinateType(value) {
  const text = String(value || 'Direct').trim()
  const lowerText = text.toLowerCase()

  if (lowerText.startsWith('direct')) {
    return 'Direct'
  }

  if (lowerText.startsWith('cart')) {
    return 'Cartesian'
  }

  return text
}

function formatInlineValue(value) {
  if (Array.isArray(value)) {
    return value.map((item) => formatInlineValue(item)).join(' ')
  }

  if (isPlainRecord(value)) {
    return JSON.stringify(value)
  }

  return String(value ?? '').trim()
}

function formatDisplayValue(value) {
  if (value === undefined || value === null) {
    return '-'
  }

  if (typeof value === 'string') {
    return value.trim() || '-'
  }

  if (Array.isArray(value)) {
    return value.map((item) => formatDisplayValue(item)).join(', ')
  }

  if (isPlainRecord(value)) {
    return JSON.stringify(value)
  }

  return String(value)
}

function normalizeStepType(type) {
  return Object.prototype.hasOwnProperty.call(workflowTypeLabels, type) ? type : 'abacus'
}

function resolveWorkflowTypes(definition) {
  const steps = Array.isArray(definition?.steps) ? definition.steps : []
  const types = steps.map((step) => step?.type).filter((type) => Object.prototype.hasOwnProperty.call(workflowTypeLabels, type))

  return [...new Set(types)]
}

function formatDependencies(step) {
  const dependencies = getStepDependencies(step)
  return dependencies.length ? dependencies.join(', ') : '-'
}

function formatStepWorkdir(step) {
  const workdir = String(
    step?.workdir ??
    step?.workDir ??
    step?.work_dir ??
    step?.working_directory ??
    step?.config?.workdir ??
    step?.config?.workDir ??
    step?.config?.work_dir ??
    step?.config?.working_directory ??
    '',
  ).trim()
  return workdir || '-'
}

function getStepDependencies(step) {
  const value = step?.depends_on ?? step?.dependencies ?? step?.dependsOn

  if (Array.isArray(value)) {
    return value.map((item) => String(item || '').trim()).filter(Boolean)
  }

  if (typeof value === 'string') {
    return value.split(',').map((item) => item.trim()).filter(Boolean)
  }

  return []
}

function isPlainRecord(value) {
  return value !== null && typeof value === 'object' && !Array.isArray(value)
}

watch(templateId, loadTemplate, { immediate: true })
</script>
