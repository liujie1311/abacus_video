<template>
  <div class="page-stack workflow-page">
    <div class="page-header">
      <div>
        <h1 class="page-title">工作流</h1>
      </div>
    </div>

    <div class="workflow-library">
      <section class="workflow-filter-panel">
        <div class="filter-main">
          <el-input
            v-model="workflowKeyword"
            :prefix-icon="Search"
            clearable
            placeholder="搜索工作流名称或 ID"
          />
          <el-input
            v-model="workflowStore.templateIdFilter"
            clearable
            placeholder="模板编号"
            @clear="applyTemplateFilterChange"
            @input="scheduleTemplateFilterChange"
            @keyup.enter="applyTemplateFilterChange"
          />
          <el-select
            v-model="workflowStore.statusFilter"
            clearable
            placeholder="按状态筛选"
            @change="handleStatusFilterChange"
          >
            <el-option v-for="status in statuses" :key="status" :label="workflowStatusText(status)" :value="status" />
          </el-select>
        </div>
        <div class="workflow-filter-actions">
          <el-button v-if="hasActiveFilters" plain @click="resetWorkflowFilters">清空筛选</el-button>
          <el-button :icon="Refresh" @click="reloadWorkflows">刷新</el-button>
        </div>
      </section>

      <div class="workflow-browser">
        <section class="workflow-list-panel" v-loading="workflowStore.loading">
          <div class="panel-head">
            <div>
              <h2>工作流列表</h2>
              <p>共 {{ workflowStore.pagination.total }} 条，当前显示 {{ filteredWorkflows.length }} 条</p>
            </div>
          </div>

          <div class="workflow-list-scroll">
            <div v-if="filteredWorkflows.length" class="workflow-card-list">
              <article
                v-for="workflow in filteredWorkflows"
                :key="workflow.id"
                class="workflow-card"
                :class="[`status-${normalizeStatus(workflow.status)}`, { active: workflow.id === workflowStore.current?.id }]"
                @click="selectWorkflow(workflow)"
              >
                <div class="workflow-card-head">
                  <span class="workflow-state-emblem">
                    <el-icon>
                      <component :is="getStatusIcon(workflow.status)" />
                    </el-icon>
                  </span>
                  <div class="workflow-card-copy">
                    <div class="workflow-card-title-row">
                      <h3 :data-i18n-skip="workflow.name ? true : undefined">{{ workflow.name || '未命名工作流' }}</h3>
                    </div>
                    <div class="workflow-id-text">
                      <span>ID</span>
                      <strong>{{ workflow.id }}</strong>
                    </div>
                  </div>
                  <div class="workflow-state-copy">
                    <strong>{{ workflowStatusText(workflow.status) }}</strong>
                  </div>
                </div>

                <div class="workflow-card-meta">
                  <span v-for="entry in getWorkflowListTimeEntries(workflow)" :key="entry.key">
                    <el-icon><Clock /></el-icon>
                    {{ entry.label }} {{ formatDate(entry.value) }}
                  </span>
                </div>

                <div class="workflow-card-actions">
                  <el-button class="card-action-button primary-action" :icon="View" @click.stop="openWorkflowDetail(workflow)">任务详情</el-button>
                  <el-button
                    v-if="canShowRetryWorkflow(workflow)"
                    class="card-action-button accent-action"
                    :disabled="!canRetryWorkflow(workflow)"
                    :icon="RefreshRight"
                    @click.stop="retry(workflow)"
                  >
                    重试
                  </el-button>
                  <el-button
                    class="card-action-button neutral-action"
                    :disabled="!canStopWorkflow(workflow)"
                    :icon="VideoPause"
                    @click.stop="stop(workflow)"
                  >
                    停止
                  </el-button>
                  <el-button
                    class="card-action-button danger-action"
                    :disabled="!canDeleteWorkflow(workflow)"
                    :icon="Delete"
                    @click.stop="remove(workflow)"
                  >
                    删除
                  </el-button>
                </div>
              </article>
            </div>
            <div v-else class="empty-block">没有找到匹配的工作流</div>
          </div>

          <div class="pagination-row">
            <el-pagination
              background
              layout="sizes, prev, pager, next, jumper"
              :current-page="workflowStore.currentPage"
              :page-size="workflowStore.pagination.limit"
              :page-sizes="[20, 50, 100, 200]"
              :total="workflowStore.pagination.total"
              @current-change="changeWorkflowPage"
              @size-change="changeWorkflowPageSize"
            />
          </div>
        </section>

        <section class="workflow-detail-panel" v-loading="workflowStore.detailLoading">
          <div v-if="!workflowStore.current" class="empty-block">请选择一个工作流</div>
          <div v-else class="workflow-detail-scroll">
            <div class="detail-hero" :class="`status-${normalizeStatus(workflowStore.current.status)}`">
              <div class="detail-hero-main">
                <span class="detail-state-emblem" :title="workflowStatusText(workflowStore.current.status)">
                  <el-icon>
                    <component :is="getStatusIcon(workflowStore.current.status)" />
                  </el-icon>
                </span>
                <div class="detail-hero-copy">
                  <h2>{{ workflowStore.current.name || '未命名工作流' }}</h2>
                </div>
              </div>
            </div>

            <div
              v-if="workflowStore.current.error"
              class="workflow-error-alert"
              role="alert"
            >
              <span class="workflow-error-icon">
                <el-icon><CircleCloseFilled /></el-icon>
              </span>
              <div class="workflow-error-copy">
                <strong>工作流错误</strong>
                <p data-i18n-skip>{{ workflowStore.current.error }}</p>
              </div>
            </div>

            <section class="workflow-section">
              <div class="section-bar">
                <div>
                  <h3 class="section-heading">任务步骤</h3>
                  <p>{{ workflowStore.tasks.length }} 个任务。</p>
                </div>
              </div>
              <div v-if="workflowStore.tasks.length" class="task-step-list">
                <article
                  v-for="(task, index) in workflowStore.tasks"
                  :key="getTaskId(task) || `${getTaskStepId(task)}-${index}`"
                  class="task-step-card"
                  :class="`status-${normalizeStatus(task.status)}`"
                >
                  <span class="task-index">{{ index + 1 }}</span>
                  <span class="task-status-emblem" :title="workflowStatusText(task.status)">
                    <el-icon>
                      <component :is="getStatusIcon(task.status)" />
                    </el-icon>
                  </span>
                  <div class="task-step-copy">
                    <div class="task-title-row">
                      <strong>{{ getTaskStepId(task) }}</strong>
                      <span v-if="getTaskType(task) !== '-'" class="task-type-tag">{{ getTaskType(task) }}</span>
                    </div>
                    <div v-if="getTaskRetryCount(task) > 0" class="task-meta-row">
                      <span v-if="getTaskRetryCount(task) > 0">重试 {{ getTaskRetryCount(task) }}</span>
                    </div>
                    <div v-if="getTaskId(task)" class="task-id-line" :title="getTaskId(task)">
                      <span class="task-id-label">ID:</span>
                      <span class="task-id-value">{{ getTaskId(task) }}</span>
                      <el-button
                        link
                        type="primary"
                        size="small"
                        :icon="CopyDocument"
                        class="task-inline-copy"
                        @click.stop="copyTaskId(task)"
                      >
                        复制
                      </el-button>
                    </div>
                    <p v-if="task.error" class="task-error" data-i18n-skip>{{ task.error }}</p>
                  </div>
                  <div v-if="getTaskCopyablePaths(task).length" class="task-output-list">
                    <div v-for="entry in getTaskCopyablePaths(task)" :key="entry.key" class="task-output-row" :title="entry.path">
                      <span class="task-output-label">{{ entry.label }}</span>
                      <span class="task-output-path" :title="entry.path" data-i18n-skip>{{ entry.path }}</span>
                      <el-button link type="primary" :icon="CopyDocument" @click.stop="copyTaskPath(entry.path)">复制</el-button>
                    </div>
                  </div>
                </article>
              </div>
              <div v-else class="empty-block compact">暂无任务步骤</div>
            </section>

          </div>
        </section>
      </div>
    </div>
  </div>
</template>

<script setup>
import { computed, onBeforeUnmount, onMounted, ref, watch } from 'vue'
import { useRoute, useRouter } from 'vue-router'
import { ElMessage, ElMessageBox } from 'element-plus'
import {
  Clock,
  CircleCheckFilled,
  CircleCloseFilled,
  CopyDocument,
  Delete,
  Loading,
  Refresh,
  RefreshRight,
  Search,
  VideoPause,
  View,
  WarningFilled,
} from '@element-plus/icons-vue'

import { useEventStore } from '@/stores/events'
import { useWorkflowStore } from '@/stores/workflows'
import { copyTextToClipboard } from '@/utils/clipboard'
import { formatDate, workflowStatusText } from '@/utils/format'

const route = useRoute()
const router = useRouter()
const eventStore = useEventStore()
const workflowStore = useWorkflowStore()
const workflowKeyword = ref('')
const statuses = ['pending', 'running', 'paused', 'failed', 'completed', 'cancelled']
const statusIcons = {
  cancelled: CircleCloseFilled,
  completed: CircleCheckFilled,
  failed: WarningFilled,
  paused: VideoPause,
  pending: Clock,
  running: Loading,
  skipped: CircleCloseFilled,
}
const templateFilterDelay = 350
const realtimeRefreshDelay = 800
let workflowListReloadTimer = 0
let currentReloadTimer = 0
let templateFilterTimer = 0
let currentReloadWorkflowId = ''

const latestRealtimeEventId = computed(() => eventStore.latestEvent?.id || '')
const routeProjectId = computed(() => String(route.query.project_id || route.query.projectId || '').trim())
const routeTemplateId = computed(() => String(route.query.template_id || route.query.templateId || '').trim())
const routeStatus = computed(() => String(route.query.status || '').trim())
const hasActiveFilters = computed(() => (
  Boolean(
    workflowKeyword.value.trim() ||
    workflowStore.statusFilter ||
    workflowStore.projectIdFilter ||
    workflowStore.templateIdFilter
  )
))
const filteredWorkflows = computed(() => {
  const keyword = normalizeKeyword(workflowKeyword.value)

  if (!keyword) {
    return workflowStore.workflows
  }

  return workflowStore.workflows.filter((workflow) => getWorkflowSearchText(workflow).includes(keyword))
})

async function loadWorkflows(params = {}) {
  await workflowStore.loadWorkflows(params)
}

async function reloadWorkflows() {
  await workflowStore.loadWorkflows()
  await ensureVisibleWorkflowSelection()
}

async function handleStatusFilterChange() {
  workflowStore.pagination.offset = 0
  router.replace({
    path: '/workflows',
    query: buildWorkflowQuery({ includeSelection: false }),
  })
  await workflowStore.loadWorkflows({ offset: 0 })
  await ensureVisibleWorkflowSelection()
}

function scheduleTemplateFilterChange() {
  window.clearTimeout(templateFilterTimer)
  templateFilterTimer = window.setTimeout(() => {
    templateFilterTimer = 0
    applyTemplateFilterChange()
  }, templateFilterDelay)
}

async function applyTemplateFilterChange() {
  window.clearTimeout(templateFilterTimer)
  templateFilterTimer = 0
  workflowStore.templateIdFilter = String(workflowStore.templateIdFilter || '').trim()
  workflowStore.projectIdFilter = ''
  workflowStore.pagination.offset = 0
  router.replace({
    path: '/workflows',
    query: buildWorkflowQuery({ includeSelection: false }),
  })
  await workflowStore.loadWorkflows({ offset: 0 })
  await ensureVisibleWorkflowSelection()
}

async function changeWorkflowPage(page) {
  await workflowStore.setPage(page)
  await ensureVisibleWorkflowSelection()
}

async function changeWorkflowPageSize(size) {
  await workflowStore.setPageSize(size)
  await ensureVisibleWorkflowSelection()
}

function openWorkflowDetail(workflow) {
  if (!workflow?.id) {
    ElMessage.warning('工作流 ID 缺失，无法查看详情')
    return
  }

  router.push({
    path: `/workflows/${encodeURIComponent(workflow.id)}`,
    query: buildWorkflowQuery({ includeSelection: false }),
  })
}

async function resetWorkflowFilters() {
  window.clearTimeout(templateFilterTimer)
  templateFilterTimer = 0
  workflowKeyword.value = ''
  workflowStore.resetFilters()
  router.replace({
    path: '/workflows',
    query: buildWorkflowQuery({ includeSelection: false }),
  })
  await workflowStore.loadWorkflows({ offset: 0 })
  await ensureVisibleWorkflowSelection()
}

function resetWorkflowPageState() {
  workflowKeyword.value = ''
  workflowStore.resetFilters()
}

function setWorkflowRouteFilters() {
  workflowStore.statusFilter = statuses.includes(routeStatus.value) ? routeStatus.value : ''
  workflowStore.projectIdFilter = routeProjectId.value
  workflowStore.templateIdFilter = routeTemplateId.value
  workflowStore.pagination.offset = 0
}

function buildWorkflowQuery({ id = '', includeSelection = true, includeTemplate = true } = {}) {
  return {
    ...(includeSelection && id ? { id } : {}),
    ...(workflowStore.statusFilter ? { status: workflowStore.statusFilter } : {}),
    ...(workflowStore.projectIdFilter ? { project_id: workflowStore.projectIdFilter } : {}),
    ...(includeTemplate && workflowStore.templateIdFilter ? { template_id: workflowStore.templateIdFilter } : {}),
  }
}

async function ensureVisibleWorkflowSelection() {
  const visibleWorkflows = filteredWorkflows.value

  if (!visibleWorkflows.length) {
    workflowStore.current = null
    workflowStore.tasks = []
    return
  }

  if (workflowStore.current?.id && visibleWorkflows.some((workflow) => workflow.id === workflowStore.current.id)) {
    return
  }

  await selectWorkflow(visibleWorkflows[0])
}

async function selectWorkflow(row) {
  if (!row?.id) {
    return
  }

  await workflowStore.selectWorkflow(row.id)
  router.replace({
    path: '/workflows',
    query: buildWorkflowQuery({ id: row.id }),
  })
}

async function retry(row) {
  if (!canRetryWorkflow(row)) {
    return
  }

  await workflowStore.retry(row.id)
  ElMessage.success('已提交重试')
}

async function stop(row) {
  if (!canStopWorkflow(row)) {
    return
  }

  const confirmed = await ElMessageBox.confirm(`确认停止工作流「${row.name}」？`, '停止工作流', {
    type: 'warning',
  })
    .then(() => true)
    .catch(() => false)

  if (!confirmed) {
    return
  }

  await workflowStore.stop(row.id)
  ElMessage.success('已停止')
}

async function remove(row) {
  if (!row?.id) {
    return
  }

  if (!canDeleteWorkflow(row)) {
    ElMessage.warning('运行中的工作流不能删除，请先停止后再删除')
    return
  }

  const confirmed = await ElMessageBox.confirm(`确认删除工作流「${row.name}」？`, '删除工作流', {
    type: 'warning',
  })
    .then(() => true)
    .catch(() => false)

  if (!confirmed) {
    return
  }

  const removedCurrent = workflowStore.current?.id === row.id

  await workflowStore.remove(row.id)

  if (removedCurrent) {
    router.replace({
      path: '/workflows',
      query: buildWorkflowQuery({ includeSelection: false }),
    })
    await ensureVisibleWorkflowSelection()
  }

  ElMessage.success('已删除')
}

function normalizeKeyword(value) {
  return String(value || '').trim().toLowerCase()
}

function normalizeStatus(status) {
  return String(status || 'unknown')
    .trim()
    .toLowerCase()
    .replace(/[^a-z0-9_-]+/g, '-')
}

function getWorkflowSearchText(workflow) {
  return normalizeKeyword(
    [
      workflow?.id,
      workflow?.name,
      workflow?.status,
      workflow?.error,
      workflow?.created_at,
      workflow?.updated_at,
    ]
      .filter(Boolean)
      .join(' '),
  )
}

function getStatusIcon(status) {
  return statusIcons[normalizeStatus(status)] || Clock
}

function getWorkflowListTimeEntries(workflow) {
  return [
    { key: 'created_at', label: '创建', value: workflow?.created_at },
    { key: 'started_at', label: '开始', value: workflow?.started_at },
    { key: 'updated_at', label: '更新时间', value: workflow?.updated_at },
  ]
}

function getWorkflowTimeEntries(workflow) {
  if (!workflow) {
    return []
  }

  const status = normalizeStatus(workflow.status)
  const finalTime =
    workflow.completed_at ||
    workflow.finished_at ||
    workflow.ended_at ||
    workflow.end_time ||
    (['completed', 'failed', 'cancelled'].includes(status) ? workflow.updated_at : '')
  const entries = [
    { key: 'created_at', label: '创建', value: workflow.created_at },
    { key: 'started_at', label: '开始', value: workflow.started_at },
    { key: 'completed_at', label: getFinishTimeLabel(status), value: finalTime },
  ]

  if (!finalTime && workflow.updated_at && workflow.updated_at !== workflow.created_at) {
    entries.push({ key: 'updated_at', label: '更新', value: workflow.updated_at })
  }

  return entries.filter((entry) => Boolean(entry.value))
}

function getFinishTimeLabel(status) {
  if (status === 'failed') {
    return '失败'
  }

  if (status === 'cancelled') {
    return '取消'
  }

  return '完成'
}

function canShowRetryWorkflow(workflow) {
  return normalizeStatus(workflow?.status) !== 'completed'
}

function canRetryWorkflow(workflow) {
  return Boolean(workflow?.id) && normalizeStatus(workflow?.status) !== 'running' && canShowRetryWorkflow(workflow)
}

function canStopWorkflow(workflow) {
  return normalizeStatus(workflow?.status) === 'running'
}

function canDeleteWorkflow(workflow) {
  return Boolean(workflow?.id) && normalizeStatus(workflow?.status) !== 'running'
}

function getTaskId(task) {
  return task?.id || task?.task_id || ''
}

function getTaskStepId(task) {
  return task?.step_id || task?.stepId || getTaskId(task) || '-'
}

function getTaskType(task) {
  return task?.type || task?.task_type || '-'
}

function getTaskRetryCount(task) {
  return Math.max(0, Number(task?.retry_count || task?.retryCount || task?.retries || 0) || 0)
}

function getTaskTimeEntries(task) {
  if (!task) {
    return []
  }

  const status = normalizeStatus(task.status)
  const finalTime =
    task.completed_at ||
    task.finished_at ||
    task.ended_at ||
    task.end_time ||
    (['completed', 'failed', 'cancelled'].includes(status) ? task.updated_at : '')
  const entries = [
    { key: 'created_at', label: '创建', value: task.created_at },
    { key: 'started_at', label: '开始', value: task.started_at },
    { key: 'completed_at', label: getFinishTimeLabel(status), value: finalTime },
  ]

  if (!finalTime && task.updated_at && task.updated_at !== task.created_at) {
    entries.push({ key: 'updated_at', label: '更新', value: task.updated_at })
  }

  return entries.filter((entry) => Boolean(entry.value))
}

function getTaskCopyablePaths(task) {
  const outputs = task?.outputs && typeof task.outputs === 'object' ? task.outputs : {}
  return ['workdir', 'output_dir']
    .map((key) => ({
      key,
      label: getTaskPathLabel(key),
      path: String(outputs[key] || '').trim(),
    }))
    .filter((entry) => isPathLikeValue(entry.path))
}

function getTaskPathLabel(key) {
  const labels = {
    workdir: '工作目录',
    output_dir: '输出目录',
  }

  return labels[key] || key
}

async function copyTaskPath(path) {
  if (!path) {
    ElMessage.warning('该任务暂无可复制的目录')
    return
  }

  try {
    await copyTextToClipboard(path)
    ElMessage.success('路径已复制')
  } catch {
    ElMessage.error('复制失败，请手动选择内容复制')
  }
}

async function copyTaskId(task) {
  const taskId = getTaskId(task)

  if (!taskId) {
    ElMessage.warning('该任务暂无可复制的 ID')
    return
  }

  try {
    await copyTextToClipboard(taskId)
    ElMessage.success('任务 ID 已复制')
  } catch {
    ElMessage.error('复制失败，请手动选择内容复制')
  }
}

function isPathLikeValue(value) {
  if (typeof value !== 'string') {
    return false
  }

  const text = value.trim()
  return /^([A-Za-z]:[\\/]|\/|\\\\)/.test(text)
}

function scheduleWorkflowListReload() {
  if (workflowListReloadTimer) {
    return
  }

  workflowListReloadTimer = window.setTimeout(async () => {
    workflowListReloadTimer = 0

    try {
      await workflowStore.loadWorkflows()
    } catch {
      // 请求拦截器会统一展示错误，这里只避免定时任务产生未处理异常。
    }
  }, realtimeRefreshDelay)
}

function scheduleCurrentReload(workflowId) {
  if (!workflowId || workflowStore.current?.id !== workflowId) {
    return
  }

  currentReloadWorkflowId = workflowId

  if (currentReloadTimer) {
    return
  }

  currentReloadTimer = window.setTimeout(async () => {
    const reloadWorkflowId = currentReloadWorkflowId

    currentReloadTimer = 0
    currentReloadWorkflowId = ''

    if (workflowStore.current?.id !== reloadWorkflowId) {
      return
    }

    try {
      await workflowStore.reloadCurrent()
    } catch {
      // 请求拦截器会统一展示错误，这里只避免定时任务产生未处理异常。
    }
  }, realtimeRefreshDelay)
}

function handleRealtimeEvent(event) {
  const result = workflowStore.applyRealtimeEvent(event)

  if (!result?.workflowId) {
    return
  }

  if (result.needsListReload) {
    scheduleWorkflowListReload()
  }

  if (!result.touchedCurrent) {
    return
  }

  scheduleCurrentReload(result.workflowId)
}

async function syncRouteSelection() {
  const routeWorkflowId = String(route.query.id || '').trim()
  const filtersChanged = (
    workflowStore.statusFilter !== (statuses.includes(routeStatus.value) ? routeStatus.value : '') ||
    workflowStore.projectIdFilter !== routeProjectId.value ||
    workflowStore.templateIdFilter !== routeTemplateId.value
  )

  if (filtersChanged) {
    setWorkflowRouteFilters()
    await workflowStore.loadWorkflows({ offset: 0 })
  }

  if (routeWorkflowId && routeWorkflowId !== workflowStore.current?.id) {
    try {
      await workflowStore.selectWorkflow(routeWorkflowId)
    } catch {
      router.replace({
        path: '/workflows',
        query: buildWorkflowQuery({ includeSelection: false }),
      })
    }
    return
  }

  if (routeWorkflowId) {
    return
  }

  if (!workflowStore.workflows.length) {
    await workflowStore.loadWorkflows()
  }

  await ensureVisibleWorkflowSelection()
}

onMounted(async () => {
  eventStore.connect()
  resetWorkflowPageState()
  setWorkflowRouteFilters()
  await loadWorkflows()
  await syncRouteSelection()

  if (!route.query.id && !workflowStore.current) {
    await ensureVisibleWorkflowSelection()
  }
})

watch(() => route.query.id, syncRouteSelection)
watch(routeStatus, syncRouteSelection)
watch(routeProjectId, syncRouteSelection)
watch(routeTemplateId, syncRouteSelection)
watch(latestRealtimeEventId, () => handleRealtimeEvent(eventStore.latestEvent))

onBeforeUnmount(() => {
  window.clearTimeout(workflowListReloadTimer)
  window.clearTimeout(currentReloadTimer)
  window.clearTimeout(templateFilterTimer)
})
</script>
